    style: 'currency',
    currency: 'MZN',
    minimumFractionDigits: 2

};

const ProtectedRoute: React.FC<{ children: React.ReactNode; requiresSuperAdmin?: boolean }> = ({ children, requiresSuperAdmin = false }) => {
  const { user, loading } = useAuth();

  if (loading) {
    return (
      <div className="flex items-center justify-center min-h-screen bg-slate-50 text-slate-900">
        <p className="text-xs text-slate-600 font-medium animate-pulse">A verificar credenciais...</p>
      </div>
    );
  }

  if (!user) {
    return <Navigate to="/login" replace />;
  }

  if (requiresSuperAdmin && user.role !== 'super_admin') {
    return <Navigate to="/app/assets" replace />;
  }

  return <>{children}</>;
};

// Componente para proteger estritamente o módulo de Gestão de Utilizadores para super_admin
const SuperAdminUsersGuard: React.FC = () => {
  const { user, loading } = useAuth();

  if (loading) return null;

  if (user?.role !== 'super_admin') {
    return <Navigate to="/app/assets" replace />;
  }

  return <UsersPage />;
};

export default function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          {/* Rota Pública */}
          <Route path="/login" element={<LoginPage />} />

          {/* Atalho direto para /users */}
          <Route path="/users" element={<Navigate to="/app/users" replace />} />

          {/* Rotas Privadas e com Layout */}
          <Route path="/" element={<Navigate to="/app/assets" replace />} />
          
          {/* Rotas de Aplicação Comum (Clientes/Gestores/Mecânicos) */}
          <Route 
            path="/app" 
            element={
              <ProtectedRoute>
                <Layout />
              </ProtectedRoute>
            }
          >
            <Route index element={<Navigate to="/app/assets" replace />} />
            <Route path="assets" element={<AssetsPage />} />
            <Route path="work-orders" element={<WorkOrdersPage />} />
            <Route path="fuel" element={<FuelPage />} />
            <Route path="inventory" element={<InventoryPage />} />
            <Route path="users" element={<SuperAdminUsersGuard />} />
            <Route path="profile" element={<ProfilePage />} />
            <Route path="settings" element={<SettingsPage />} />
          </Route>

          {/* Rota Privada Super Admin */}
          <Route 
            path="/super-admin" 
            element={
              <ProtectedRoute requiresSuperAdmin>
                <Layout />
              </ProtectedRoute>
            }
          >
            <Route index element={<SuperAdminPage />} />
          </Route>

          {/* Rota de Fallback */}
          <Route path="*" element={<Navigate to="/login" replace />} />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  );
}

const formatTimeAgo = (dateStr?: string) => {
  if (!dateStr) return '';
  const date = new Date(dateStr);
  if (isNaN(date.getTime())) return dateStr;
  
  const now = new Date();
  const diffInSeconds = Math.floor((now.getTime() - date.getTime()) / 1000);
  
  if (diffInSeconds < 60) return 'Agora mesmo';
  const diffInMinutes = Math.floor(diffInSeconds / 60);
  if (diffInMinutes < 60) return `Há ${diffInMinutes} min`;
  const diffInHours = Math.floor(diffInMinutes / 60);
  if (diffInHours < 24) return `Há ${diffInHours}h`;
  const diffInDays = Math.floor(diffInHours / 24);
  if (diffInDays < 7) return `Há ${diffInDays}d`;
  
  return date.toLocaleDateString('pt-MZ', { day: '2-digit', month: '2-digit', year: 'numeric' });
};

  const notificationRef = useRef<HTMLDivElement>(null);

  // Busca notificações na API (GET /notifications)
  const fetchNotifications = async () => {
    try {
      const response = await api.get('/notifications');
      const data = response.data;
      let list: NotificationItem[] = [];
      let count = 0;

      if (Array.isArray(data)) {
        list = data;
        count = list.filter(n => !(n.is_read ?? n.read ?? n.isRead)).length;
      } else if (data && typeof data === 'object') {
        const possibleList = data.notifications || data.data || data.items || [];
        if (Array.isArray(possibleList)) {
          list = possibleList;
        }
        if (typeof data.unreadCount === 'number') {
          count = data.unreadCount;
        } else if (typeof data.unread_count === 'number') {
          count = data.unread_count;
        } else {
          count = list.filter(n => !(n.is_read ?? n.read ?? n.isRead)).length;
        }
      }

      setNotifications(list);
      setUnreadCount(count);
    } catch (err) {
      console.warn('Não foi possível obter as notificações:', err);
    }
  };

  useEffect(() => {
    if (user) {
      fetchNotifications();
      const interval = setInterval(fetchNotifications, 60000);
      return () => clearInterval(interval);
    }
  }, [user]);

  // Fechar dropdown de notificações ao clicar fora
  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (notificationRef.current && !notificationRef.current.contains(event.target as Node)) {
        setIsNotificationOpen(false);
      }
    };
    if (isNotificationOpen) {
      document.addEventListener('mousedown', handleClickOutside);

    const isRead = Boolean(item.is_read ?? item.read ?? item.isRead);
    const itemLink = item.link || item.targetUrl || item.target_url || item.url;

    if (!isRead) {
      try {
        await api.patch(`/notifications/${item.id}/read`);
      } catch (err) {
        console.warn(`Erro ao marcar notificação ${item.id} como lida:`, err);
      } finally {
        setNotifications(prev =>
          prev.map(n => (n.id === item.id ? { ...n, is_read: true, read: true, isRead: true } : n))

import React, { createContext, useContext, useState, useEffect } from 'react';
import { api } from '../services/api';

export type UserRole = 'super_admin' | 'tenant_admin' | 'supervisor' | 'technician' | 'driver_operator';

export interface User {
  id: string;
  fullName: string;
  email: string;
  role: UserRole;
  tenantName: string;
  tenantId: string;
  operational_scope?: 'fleet' | 'plant' | 'both';
  tenant?: {
    id: string;
    name: string;
    max_users?: number;
    maxUsers?: number;
    operational_scope?: 'fleet' | 'plant' | 'both';
    operationalScope?: 'fleet' | 'plant' | 'both';
  };
}

interface AuthContextType {
  user: User | null;
  loading: boolean;
  logout: () => Promise<void>;
  login: (email: string, password: string) => Promise<User>;
  setUser: React.Dispatch<React.SetStateAction<User | null>>;
  isSuperAdmin: boolean;
  isTenantAdmin: boolean;
  isSupervisor: boolean;
  isTechnician: boolean;
  isDriverOperator: boolean;
  isFleet: boolean;
  isPlant: boolean;
  currentScope: 'fleet' | 'plant';
  setCurrentScope: React.Dispatch<React.SetStateAction<'fleet' | 'plant'>>;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  const [currentScope, setCurrentScope] = useState<'fleet' | 'plant'>('fleet');

  const checkAuth = async () => {
    try {
      const response = await api.get('/auth/me');
      const userData = response.data?.user || response.data;
      if (userData && userData.id && userData.role) {
        setUser(userData);
        // Inicializar o escopo padrão do utilizador
        const scope = userData.operational_scope || userData.tenant?.operational_scope || userData.tenant?.operationalScope;
        if (scope === 'plant') {
          setCurrentScope('plant');
        } else {
          setCurrentScope('fleet');
        }
      } else {
        setUser(null);
      }
    } catch (err) {
      setUser(null);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    checkAuth();
  }, []);

  useEffect(() => {
    if (user) {
      const scope = user.operational_scope || user.tenant?.operational_scope || user.tenant?.operationalScope;
      if (scope === 'plant') {
        setCurrentScope('plant');
      } else if (scope === 'fleet') {
        setCurrentScope('fleet');
      }
    }
  }, [user]);

  const login = async (email: string, password: string): Promise<User> => {
    try {
      const response = await api.post('/auth/login', { email, password });
      const userData = response.data?.user || response.data;
      
      if (!userData || !userData.id || !userData.role) {
        throw new Error('Servidor retornou dados de utilizador inválidos.');
      }
      
      setUser(userData);
      const scope = userData.operational_scope || userData.tenant?.operational_scope || userData.tenant?.operationalScope;
      if (scope === 'plant') {
        setCurrentScope('plant');
      } else {
        setCurrentScope('fleet');
      }
      return userData;
    } catch (err: any) {
      const errorMsg = err.response?.data?.message || err.response?.data?.error || err.message || 'Erro ao efetuar login.';
      throw new Error(errorMsg);
    }
  };

  const logout = async () => {
    setLoading(true);
    try {
      await api.post('/auth/logout').catch(() => {});
    } catch (err) {
      console.error('Erro ao efetuar logout no servidor', err);
    } finally {
      setUser(null);
      setLoading(false);
    }
  };

  const isSuperAdmin = user?.role === 'super_admin';
  const isTenantAdmin = user?.role === 'tenant_admin';
  const isSupervisor = user?.role === 'supervisor';
  const isTechnician = user?.role === 'technician';
  const isDriverOperator = user?.role === 'driver_operator';

  const isFleet = currentScope === 'fleet';
  const isPlant = currentScope === 'plant';

  return (
    <AuthContext.Provider
      value={{
        user,
        loading,
        logout,
        login,
        setUser,
        isSuperAdmin,
        isTenantAdmin,
        isSupervisor,
        isTechnician,
        isDriverOperator,
        isFleet,
        isPlant,
        currentScope,
        setCurrentScope
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (context === undefined) {
    throw new Error('useAuth deve ser utilizado dentro de um AuthProvider');
  }
  return context;
};

// ===== src/hooks/usePreventRefresh.ts =====
import { useEffect } from 'react';

/**
 * Hook para previnir o recarregamento ou encerramento acidental da página
 * no navegador (F5, fechar aba, fechar janela, botão recarregar).
 *
 * @param enabled Define se a proteção contra recarregamento está ativa (padrão: true)
 */
export const usePreventRefresh = (enabled: boolean = true) => {
  useEffect(() => {
    if (!enabled) return;

    const handleBeforeUnload = (event: BeforeUnloadEvent) => {
      // Cancela o evento conforme especificação
      event.preventDefault();
      // O Chrome requer que returnValue seja definido com uma string
      event.returnValue = '';
      return '';
    };

    window.addEventListener('beforeunload', handleBeforeUnload);

    return () => {
      window.removeEventListener('beforeunload', handleBeforeUnload);
    };
  }, [enabled]);
};

export default usePreventRefresh;

      const timeStr = ev.created_at || ev.createdAt || '';

    const openedDate = new Date(order.opened_at || order.openedAt || '');
    const diffTime = Math.abs(new Date().getTime() - openedDate.getTime());
    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
    return diffDays > 3;
  };

  // Abre Drawer do Técnico
  const handleOpenDrawer = (order: WorkOrder) => {
    setSelectedDrawerOrder(order);
    setIsDrawerOpen(true);
    fetchTimeline(order.id);
    fetchUsedParts(order.id);
    fetchInventoryParts();
  };

  // Renderização das badges de prioridade
  const renderPriorityBadge = (prio: string) => {
    switch (prio) {
      case 'low':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-medium bg-slate-100 text-slate-700 border border-slate-200">
            Baixa
          </span>
        );
      case 'medium':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-medium bg-blue-50 text-blue-700 border border-blue-200">
            Média
          </span>
        );
      case 'high':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-medium bg-amber-50 text-amber-700 border border-amber-200">
            Alta
          </span>
        );
      case 'urgent':
      case 'critical':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-semibold bg-red-50 text-red-700 border border-red-200 animate-pulse">
            ⚠️ Urgente
          </span>
        );
      default:
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-medium bg-slate-100 text-slate-600 border border-slate-200">
            {prio}
          </span>
        );
    }
  };

  // Renderização das badges de status
  const renderStatusBadge = (status: string) => {
    switch (status) {
      case 'open':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-amber-50 text-amber-700 border border-amber-200">
            Aberta
          </span>
        );
      case 'in_progress':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-blue-50 text-blue-700 border border-blue-200">
            Em Curso
          </span>
        );
      case 'waiting_parts':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-purple-50 text-purple-700 border border-purple-200">
            Aguardar Peças
          </span>
        );
      case 'paused':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-slate-100 text-slate-700 border border-slate-200">
            Pausada
          </span>
        );
      case 'blocked':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-red-50 text-red-700 border border-red-200">
            Bloqueada
          </span>
        );
      case 'completed':
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-emerald-50 text-emerald-700 border border-emerald-200">
            Concluída
          </span>
        );
      default:
        return (
          <span className="inline-flex items-center gap-1 px-2.5 py-0.5 rounded-md text-[11px] font-medium bg-slate-100 text-slate-600 border border-slate-200">
            {status}
          </span>
        );
    }
  };

  // Resolução do Nome do Técnico Encarregue (Eliminar UUIDs nos Cards)
  const resolveTechnicianName = (order: any) => {
    if (!order) return 'Pendente de Atribuição';
    if (order.assigned_technician_name) return order.assigned_technician_name;
    if (order.assignedTechnicianName) return order.assignedTechnicianName;
    if (order.technician_name) return order.technician_name;
    if (order.technician?.name) return order.technician.name;
    
    // Fallback: procura na lista de técnicos/utilizadores já carregada em memória
    const matchedTech = technicians.find(t => 
      t.id === order.assigned_to || 
      t.id === order.assignedTo || 
      t.id === order.technician_id || 
      t.id === order.technicianId
    );
    if (matchedTech) return matchedTech.name || (matchedTech as any).fullName;

    // Proteção estrita: se for um UUID ou estiver vazio, NUNCA exibir o código na tela
    const isUUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
    const assignedVal = order.assigned_to || order.assignedTo;
    if (!assignedVal || isUUID.test(String(assignedVal))) {
      return 'Pendente de Atribuição';
    }
    return String(assignedVal);
  };

  // Helper de Formatação dos Contadores (KM vs Horas)
  const formatMeterReading = (val: any, unit: string) => {
    if (val === undefined || val === null || val === '') return 'Sem registo';
    const num = Number(val);
    if (isNaN(num)) return 'Sem registo';
    if (unit === 'km') {
      return `${num.toLocaleString('pt-MZ')} km`;
    }
    return `${num.toLocaleString('pt-MZ', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} horas`;
  };

  // Filtro defensivo de isolamento de domínio (fleet vs plant)
  const domainFilteredOrders = orders.filter((order) => {
    const matchingAsset = assets.find(a => a.id === order.asset_id || a.id === order.assetId);
    
    if (domain === 'plant') {
      const orderAssetType = (order as any).asset_type || (order as any).assetType;
      const orderMeterType = (order as any).meter_type || (order as any).meterType || (order as any).meter_kind || (order as any).meterKind;
      if (orderAssetType === 'fleet') return false;
      if (orderMeterType === 'kms' || orderMeterType === 'odometer_km') return false;
      if (matchingAsset) {
        const isPlantCategory = matchingAsset.category === 'generator' || matchingAsset.category === 'machinery' || (matchingAsset as any).category_type === 'plant' || (matchingAsset as any).type === 'plant';
        const isFleetCategory = matchingAsset.category === 'vehicle_light' || matchingAsset.category === 'vehicle_heavy' || (matchingAsset as any).category_type === 'fleet' || (matchingAsset as any).type === 'fleet';
        if (isFleetCategory && !isPlantCategory) return false;
      }
    } else if (domain === 'fleet') {
      const orderAssetType = (order as any).asset_type || (order as any).assetType;
      const orderMeterType = (order as any).meter_type || (order as any).meterType || (order as any).meter_kind || (order as any).meterKind;
      if (orderAssetType === 'plant') return false;
      if (orderMeterType === 'hours' || orderMeterType === 'hourmeter') return false;
      if (matchingAsset) {
        const isPlantCategory = matchingAsset.category === 'generator' || matchingAsset.category === 'machinery' || (matchingAsset as any).category_type === 'plant' || (matchingAsset as any).type === 'plant';
        const isFleetCategory = matchingAsset.category === 'vehicle_light' || matchingAsset.category === 'vehicle_heavy' || (matchingAsset as any).category_type === 'fleet' || (matchingAsset as any).type === 'fleet';
        if (isPlantCategory && !isFleetCategory) return false;
      }
    }
    return true;
  });

  // Filtragem local inteligente baseada nos critérios de pesquisa
  const filteredOrders = domainFilteredOrders.filter((order) => {
    const title = (order.title || '').toLowerCase();
    const description = (order.description || '').toLowerCase();
    const woNum = (order.wo_number || order.woNumber || '').toLowerCase();
    const assigned = resolveTechnicianName(order).toLowerCase();
    
    const matchingAsset = assets.find(a => a.id === order.asset_id || a.id === order.assetId);
    const assetString = matchingAsset 
      ? `${matchingAsset.code_tag || matchingAsset.codeTag} ${matchingAsset.name}`.toLowerCase()
      : `${order.asset?.code_tag || ''} ${order.asset_name || order.asset?.name || ''}`.toLowerCase();
    
    const query = searchQuery.toLowerCase();

    const matchesSearch = 
      title.includes(query) || 
      description.includes(query) ||
      assetString.includes(query) || 
      woNum.includes(query) ||
      assigned.includes(query);

    const matchesPriority = priorityFilter === 'all' || order.priority === priorityFilter;
    
    // Quick Tab filter (Requisito 3)
    let matchesQuickTab = true;
    if (quickTab === 'my_tasks') {
      matchesQuickTab = isAssignedToMe(order) && order.status !== 'completed';
    } else if (quickTab === 'pending') {
      matchesQuickTab = order.status === 'open';
    } else if (quickTab === 'in_progress') {
      matchesQuickTab = order.status === 'in_progress';
    } else if (quickTab === 'completed') {
      matchesQuickTab = order.status === 'completed';
    }

    // Se a tab estiver em 'all', respeitamos o select de status filter
    const matchesStatus = quickTab !== 'all' 
      ? true 
      : (statusFilter === 'all' || order.status === statusFilter);

    return matchesSearch && matchesPriority && matchesQuickTab && matchesStatus;
  });

  return (
    <div className="space-y-6">
      
      {/* Toast de Sucesso */}
      {successMessage && (
        <div className="fixed bottom-6 right-6 z-50 bg-white border border-slate-200 text-slate-900 p-4 rounded-md shadow-lg max-w-md flex gap-3 items-start animate-bounce">
          <CheckCircle2 className="text-blue-700 shrink-0 mt-0.5" size={18} />
          <div>
            <span className="font-bold block text-sm text-slate-900">Cockpit de Manutenção</span>
            <span className="text-xs text-slate-600">{successMessage}</span>
          </div>
        </div>
      )}

      {/* Cabeçalho */}
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 animate-fade-in">
        <div>
          <h1 className="text-xl md:text-2xl font-bold tracking-tight text-slate-900 flex items-center gap-2">
            <span>Ordens de Serviço</span>
            {loading && <Loader2 size={18} className="animate-spin text-slate-400" />}
          </h1>
          <p className="text-slate-500 text-xs mt-1">
            Cockpit móvel para acompanhamento de tarefas, transição de estados e linha do tempo de auditoria.
          </p>
        </div>
        <div className="flex gap-2 relative">
          
          {/* Sino de Notificações para Técnicos (Requisito 2) */}
          <div className="relative">
            <button
              onClick={() => {
                setIsBellOpen(!isBellOpen);
                fetchNotifications();
              }}
              className="p-2.5 bg-white hover:bg-slate-50 border border-slate-300 text-slate-700 rounded-md transition-all cursor-pointer relative"
              title="Notificações de Tarefas Pendentes"
            >
              <Bell size={16} />
              {notifications.length > 0 && (
                <span className="absolute -top-1 -right-1 bg-red-600 text-white text-[9px] font-bold w-4 h-4 rounded-full flex items-center justify-center animate-pulse">
                  {notifications.length}
                </span>
              )}
            </button>

            {/* Dropdown Flutuante de Notificações */}
            {isBellOpen && (
              <div className="absolute right-0 mt-2 w-72 bg-white border border-slate-200 rounded-md shadow-xl z-50 overflow-hidden animate-fade-in text-left">
                <div className="p-3.5 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
                  <span className="text-xs font-bold text-slate-900 flex items-center gap-1.5">
                    <Bell size={13} className="text-blue-700" />
                    <span>Tarefas Atribuídas ({notifications.length})</span>
                  </span>
                  <button 
                    onClick={() => setIsBellOpen(false)}
                    className="text-[10px] text-slate-500 hover:text-slate-800"
                  >
                    Fechar
                  </button>
                </div>

                <div className="max-h-[280px] overflow-y-auto divide-y divide-slate-100">
                  {loadingNotifications ? (
                    <div className="p-4 text-center">
                      <Loader2 size={16} className="animate-spin text-slate-400 mx-auto" />
                    </div>
                  ) : notifications.length === 0 ? (
                    <div className="p-6 text-center text-slate-500 text-xs italic">
                      Nenhuma tarefa pendente atribuída.
                    </div>
                  ) : (
                    notifications.map((note) => (
                      <div
                        key={note.id}
                        onClick={() => {
                          const targetOrder = note.order || orders.find(o => o.id === note.orderId || o.id === note.order_id);
                          if (targetOrder) {
                            handleOpenDrawer(targetOrder);
                          }
                          setIsBellOpen(false);
                        }}
                        className="p-3 hover:bg-slate-50 cursor-pointer transition-colors space-y-1"
                      >
                        <div className="text-[11px] font-bold text-slate-900 leading-tight flex items-center justify-between">
                          <span>{note.title}</span>
                          <span className="w-1.5 h-1.5 rounded-full bg-blue-600"></span>
                        </div>
                        <p className="text-[10px] text-slate-600 line-clamp-2 leading-relaxed">
                          {note.message}
                        </p>
                        <span className="text-[9px] text-slate-400 block font-mono">
                          {new Date(note.createdAt).toLocaleString('pt-MZ')}
                        </span>
                      </div>
                    ))
                  )}
                </div>
              </div>
            )}
          </div>

          <button
            onClick={fetchWorkOrders}
            disabled={loading}
            className="p-2.5 bg-white hover:bg-slate-50 border border-slate-300 text-slate-700 rounded-md transition-all cursor-pointer disabled:opacity-50"
            title="Recarregar ordens"
          >
            <RefreshCw size={16} className={loading ? "animate-spin" : ""} />
          </button>
          <button
            onClick={handleOpenCreateModal}
            className="bg-blue-700 hover:bg-blue-800 active:bg-blue-900 text-white text-xs font-medium py-2.5 px-4 rounded-md flex items-center gap-1.5 transition-colors cursor-pointer shadow-none"
          >
            <Plus size={16} />
            <span>Nova Ordem de Serviço</span>
          </button>
        </div>
      </div>

      {/* Erros da API */}
      {error && (
        <div className="p-4 bg-red-50 border border-red-200 text-red-700 text-xs rounded-md flex gap-3 items-start animate-fade-in">
          <AlertTriangle size={18} className="text-red-600 shrink-0 mt-0.5" />
          <div className="flex-1">
            <span className="font-bold block text-red-800 mb-0.5">Erro de Ligação</span>
            <span className="leading-relaxed block text-red-700">{error}</span>
          </div>
        </div>
      )}

      {/* Filtros e Caixa de Busca */}
      <div className="bg-white border border-slate-200 rounded-md p-4 flex flex-col md:flex-row gap-3 shadow-none">
        <div className="relative flex-1">
          <Search size={14} className="absolute left-3.5 top-3.5 text-slate-400" />
          <input
            type="text"
            placeholder="Pesquisar por número, atividade, técnico, matrícula ou tag do ativo..."
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md pl-9 pr-3 py-2 text-xs text-slate-900 placeholder-slate-400 focus:outline-none"
          />
        </div>
        <div className="flex gap-2">
          <select
            value={priorityFilter}
            onChange={(e) => setPriorityFilter(e.target.value)}
            className="bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-700 focus:outline-none"
          >
            <option value="all">Todas as Prioridades</option>
            <option value="low">Baixa</option>
            <option value="medium">Média</option>
            <option value="high">Alta</option>
            <option value="urgent">Urgente</option>
          </select>
          <select
            value={statusFilter}
            onChange={(e) => setStatusFilter(e.target.value)}
            className="bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-700 focus:outline-none"
          >
            <option value="all">Todos os Estados</option>
            <option value="open">Abertas</option>
            <option value="in_progress">Em Curso</option>
            <option value="waiting_parts">Aguardando Peças</option>
            <option value="paused">Pausadas / Interrompidas</option>
            <option value="completed">Concluídas</option>
          </select>
        </div>
      </div>

      {/* Abas/Filtros Rápidos no Topo (Requisito 3) */}
      <div className="flex overflow-x-auto gap-2 pb-2 scrollbar-none shrink-0 -mx-4 px-4 md:mx-0 md:px-0">
        <button
          onClick={() => setQuickTab('all')}
          className={`px-4 py-2 rounded-md text-xs font-medium transition-all whitespace-nowrap cursor-pointer border ${
            quickTab === 'all'
              ? 'bg-blue-700 text-white border-blue-700 shadow-none font-semibold'
              : 'bg-white text-slate-600 border-slate-300 hover:bg-slate-50'
          }`}
        >
          Todas ({domainFilteredOrders.length})
        </button>
        <button
          onClick={() => setQuickTab('my_tasks')}
          className={`px-4 py-2 rounded-md text-xs font-medium transition-all whitespace-nowrap cursor-pointer border ${
            quickTab === 'my_tasks'
              ? 'bg-blue-700 text-white border-blue-700 shadow-none font-semibold'
              : 'bg-white text-slate-600 border-slate-300 hover:bg-slate-50'
          }`}
        >
          Minhas Tarefas ({domainFilteredOrders.filter(isAssignedToMe).filter(o => o.status !== 'completed').length})
        </button>
        <button
          onClick={() => setQuickTab('pending')}
          className={`px-4 py-2 rounded-md text-xs font-medium transition-all whitespace-nowrap cursor-pointer border ${
            quickTab === 'pending'
              ? 'bg-blue-700 text-white border-blue-700 shadow-none font-semibold'
              : 'bg-white text-slate-600 border-slate-300 hover:bg-slate-50'
          }`}
        >
          Pendentes de Início ({domainFilteredOrders.filter(o => o.status === 'open').length})
        </button>
        <button
          onClick={() => setQuickTab('in_progress')}
          className={`px-4 py-2 rounded-md text-xs font-medium transition-all whitespace-nowrap cursor-pointer border ${
            quickTab === 'in_progress'
              ? 'bg-blue-700 text-white border-blue-700 shadow-none font-semibold'
              : 'bg-white text-slate-600 border-slate-300 hover:bg-slate-50'
          }`}
        >
          Em Andamento ({domainFilteredOrders.filter(o => o.status === 'in_progress').length})
        </button>
        <button
          onClick={() => setQuickTab('completed')}
          className={`px-4 py-2 rounded-md text-xs font-medium transition-all whitespace-nowrap cursor-pointer border ${
            quickTab === 'completed'
              ? 'bg-blue-700 text-white border-blue-700 shadow-none font-semibold'
              : 'bg-white text-slate-600 border-slate-300 hover:bg-slate-50'
          }`}
        >
          Concluídas ({domainFilteredOrders.filter(o => o.status === 'completed').length})
        </button>
      </div>

      {/* Grid de Ordens de Serviço (Tabelas/Cartões Reais) */}
      {loading ? (
        <div className="p-12 text-center bg-white border border-slate-200 rounded-md">
          <Loader2 className="animate-spin text-blue-700 mx-auto" size={32} />
          <p className="text-xs text-slate-500 mt-3 font-medium">A carregar registos de ordens de serviço do CMMS...</p>
        </div>
      ) : filteredOrders.length === 0 ? (
        <div className="bg-white border border-slate-200 rounded-md text-center py-20 px-4">
          <Briefcase size={36} className="text-slate-300 mx-auto mb-4" />
          <h3 className="text-slate-700 font-bold text-sm">Nenhuma Ordem de Serviço Ativa</h3>
          <p className="text-slate-500 text-xs mt-1">Crie uma nova OS ou redefina os filtros aplicados.</p>
        </div>
      ) : (
        <div className="space-y-4">
          <span className="text-xs font-semibold text-slate-500 uppercase tracking-wider block">
            Ordens Filtradas ({filteredOrders.length}) · Clique para abrir o cockpit de campo
          </span>
          
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
            {filteredOrders.map((order) => {
              const matchingAsset = assets.find(a => a.id === order.asset_id || a.id === order.assetId);
              const assetDisplay = matchingAsset 
                ? `${matchingAsset.code_tag || matchingAsset.codeTag} • ${matchingAsset.name}`
                : order.asset_name || order.asset?.name || 'Equipamento Geral';

              const priority = order.priority || 'medium';
              const isUrgent = priority === 'urgent' || priority === 'critical';
              const delayed = isDelayed(order);
              
              const techDisplay = resolveTechnicianName(order);

              const meterVal = order.meter_at_service ?? order.meterAtService;
              const orderMeterType = (order as any).meter_type || (order as any).meterType || (order as any).meter_kind || (order as any).meterKind;
              const isKm = domain === 'fleet' || orderMeterType === 'kms' || orderMeterType === 'odometer_km' || matchingAsset?.meter_kind === 'odometer_km' || matchingAsset?.meterKind === 'odometer_km';
              const meterUnit = isKm ? 'km' : 'horas';

              return (
                <div
                  key={order.id}
                  onClick={() => handleOpenDrawer(order)}
                  className={`bg-white border rounded-md p-4 space-y-3.5 cursor-pointer hover:border-slate-300 transition-all shadow-none flex flex-col justify-between ${
                    isUrgent 
                      ? 'border-red-300 hover:border-red-400 bg-red-50/20' 
                      : delayed 
                      ? 'border-amber-300 hover:border-amber-400' 
                      : 'border-slate-200 hover:border-slate-300'
                  }`}
                >
                  <div className="space-y-2.5">
                    <div className="flex justify-between items-center pb-2 border-b border-slate-100">
                      <div className="flex items-center gap-1.5 truncate">
                        <span className="font-mono font-bold text-[10px] text-slate-700 bg-slate-100 px-2 py-0.5 rounded border border-slate-200">
                          {matchingAsset ? matchingAsset.code_tag || matchingAsset.codeTag : 'EQ-GERAL'}
                        </span>
                        <span className="font-mono font-semibold text-[10px] text-blue-700 bg-blue-50 px-1.5 py-0.5 rounded shrink-0 border border-blue-100">
                          {order.wo_number || order.woNumber || `OS-${order.id.substring(0, 4)}`}
                        </span>
                      </div>
                      <div className="flex gap-1.5 items-center shrink-0">
                        {delayed && (
                          <span className="bg-amber-50 text-amber-700 text-[9px] font-bold px-1.5 py-0.5 border border-amber-200 rounded">
                            Atraso
                          </span>
                        )}
                        {renderStatusBadge(order.status)}
                      </div>
                    </div>

                    <div className="space-y-1">
                      <h3 className="text-slate-900 font-bold text-sm leading-snug truncate" title={order.title}>
                        {order.title}
                      </h3>
                      <div className="flex items-center gap-1.5 text-[10px] text-slate-500 font-mono">
                        <Activity size={11} className="text-slate-400 shrink-0" />
                        <span>Contador de Entrada:</span>
                        <span className="text-slate-800 font-semibold">
                          {formatMeterReading(meterVal, meterUnit)}
                        </span>
                      </div>
                      <p className="text-slate-600 text-xs line-clamp-2 leading-relaxed">
                        {order.description || 'Sem descrição detalhada.'}
                      </p>
                    </div>
                  </div>

                  <div className="pt-3 border-t border-slate-100 space-y-3 mt-2">
                    <div className="flex justify-between items-end">
                      <div>
                        <span className="text-[9px] text-slate-500 uppercase font-bold tracking-wide">Técnico Encarregue</span>
                        <span className="text-slate-800 text-xs font-semibold block truncate max-w-[150px]">
                          {techDisplay}
                        </span>
                      </div>
                      <div className="flex items-center gap-1.5">
                        {renderPriorityBadge(priority)}
                        <button
                          onClick={(e) => handleOpenEditModal(order, e)}
                          className="p-1 text-slate-400 hover:text-slate-700 hover:bg-slate-100 rounded transition-all cursor-pointer"
                          title="Editar Ficha"
                        >
                          <Edit2 size={13} />
                        </button>
                      </div>
                    </div>

                    {/* Botões de Ação Direta no próprio cartão (Requisito 1) */}
                    <div className="pt-1.5 border-t border-slate-100" onClick={(e) => e.stopPropagation()}>
                      {order.status === 'open' && (
                        <button
                          onClick={(e) => {
                            e.stopPropagation();
                            handleStartService(order);
                          }}
                          className="w-full py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all shadow-none cursor-pointer"
                        >
                          <Play size={12} className="fill-current shrink-0" />
                          <span>▶ Iniciar Serviço</span>
                        </button>
                      )}

                      {order.status === 'in_progress' && (
                        <div className="grid grid-cols-2 gap-2">
                          <button
                            onClick={(e) => {
                              e.stopPropagation();
                              handleOpenPauseModal(order);
                            }}
                            className="py-2 bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all cursor-pointer"
                          >
                            <Pause size={12} className="shrink-0" />
                            <span>⏸ Pausar</span>
                          </button>
                          <button
                            onClick={(e) => {
                              e.stopPropagation();
                              setSelectedOrder(order);
                              setIsCloseModalOpen(true);
                            }}
                            className="py-2 bg-emerald-700 hover:bg-emerald-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all cursor-pointer shadow-none"
                          >
                            <CheckCircle2 size={12} className="shrink-0" />
                            <span>✔ Concluir</span>
                          </button>
                        </div>
                      )}

                      {(order.status === 'waiting_parts' || order.status === 'paused' || order.status === 'blocked') && (
                        <button
                          onClick={(e) => {
                            e.stopPropagation();
                            handleResumeService(order);
                          }}
                          className="w-full py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all shadow-none cursor-pointer"
                        >
                          <Play size={12} className="fill-current shrink-0" />
                          <span>▶ Retomar</span>
                        </button>
                      )}
                    </div>
                  </div>

                </div>
              );
            })}
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: CRIAR OU EDITAR OS (COM TÉCNICOS INTEGRADOS)
          ========================================== */}
      {isFormModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-lg rounded-md shadow-xl overflow-hidden animate-slide-up">
            
            <div className="p-5 border-b border-slate-200 flex items-center justify-between bg-white">
              <div className="flex items-center gap-2">
                <Wrench className="text-blue-700" size={18} />
                <h2 className="text-base font-bold text-slate-900">
                  {formMode === 'create' ? 'Abrir Ordem de Serviço' : 'Editar Ficha da OS'}
                </h2>
              </div>
              <button
                onClick={() => setIsFormModalOpen(false)}
                className="p-1.5 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            <form onSubmit={handleSaveOrder} className="p-6 space-y-4 max-h-[75vh] overflow-y-auto">
              
              <div>
                <label htmlFor="formAsset" className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                  Ativo / Equipamento Beneficiário *
                </label>
                <select
                  id="formAsset"
                  required
                  disabled={formMode === 'edit'}
                  value={formData.assetId || formData.asset_id || ''}
                  onChange={(e) => handleAssetChange(e.target.value)}
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 cursor-pointer focus:outline-none disabled:bg-slate-50 disabled:text-slate-500"
                >
                  <option value="">Selecione o Ativo...</option>
                  {domainAssets.map((asset) => (
                    <option key={asset.id} value={asset.id}>
                      [{asset.code_tag || asset.codeTag}] {asset.name}
                    </option>
                  ))}
                </select>
              </div>

              <div>
                <div className="flex items-center justify-between mb-1">
                  <label htmlFor="formTitle" className="block text-xs font-semibold text-slate-600 uppercase tracking-wide">
                    Título da Avaria / Intervenção *
                  </label>
                </div>
                <select
                  id="formCause"
                  onChange={(e) => {
                    if (e.target.value) {
                      const found = failureCauses.find(fc => fc.value === e.target.value);
                      if (found) {
                        setFormData((prev: any) => ({ ...prev, title: found.label }));
                      }
                    }
                  }}
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-1.5 text-xs text-slate-700 cursor-pointer focus:outline-none mb-2"
                >
                  <option value="">Selecione a Causa da Avaria / Intervenção...</option>
                  {failureCauses.map((fc) => (
                    <option key={fc.value} value={fc.value}>
                      {fc.label}
                    </option>
                  ))}
                </select>
                <input
                  id="formTitle"
                  type="text"
                  required
                  value={formData.title || ''}
                  onChange={(e) => setFormData({ ...formData, title: e.target.value })}
                  placeholder="ou digite o título da avaria..."
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none placeholder-slate-400"
                />
              </div>

              <div>
                <label htmlFor="formDesc" className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                  Descrição Técnica do Problema ou Intervenção *
                </label>
                <textarea
                  id="formDesc"
                  required
                  rows={3}
                  value={formData.description || ''}
                  onChange={(e) => setFormData({ ...formData, description: e.target.value })}
                  placeholder="Descreva detalhadamente o serviço técnico ou anomalia operacional..."
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none placeholder-slate-400"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label htmlFor="formType" className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                    Tipo de Manutenção
                  </label>
                  <select
                    id="formType"
                    value={formData.orderType || formData.order_type || 'corrective'}
                    onChange={(e) => setFormData({ ...formData, orderType: e.target.value as any, order_type: e.target.value as any })}
                    className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 cursor-pointer focus:outline-none"
                  >
                    <option value="corrective">Corretiva (Avaria/Paragem)</option>
                    <option value="preventive">Preventiva (Revisão/Plano)</option>
                  </select>
                </div>
                <div>
                  <label htmlFor="formPriority" className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                    Nível de Urgência / Prioridade
                  </label>
                  <select
                    id="formPriority"
                    value={formData.priority || 'medium'}
                    onChange={(e) => setFormData({ ...formData, priority: e.target.value as any })}
                    className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 cursor-pointer focus:outline-none"
                  >
                    <option value="low">Baixa</option>
                    <option value="medium">Média</option>
                    <option value="high">Alta</option>
                    <option value="urgent">Urgente (Imediato)</option>
                  </select>
                </div>
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label htmlFor="formMeter" className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                    {counterLabel}
                  </label>
                  <input
                    id="formMeter"
                    type="number"
                    min={0}
                    value={formData.meterAtService ?? formData.meter_at_service ?? ''}
                    onChange={(e) => setFormData({ ...formData, meterAtService: e.target.value !== '' ? Number(e.target.value) : '', meter_at_service: e.target.value !== '' ? Number(e.target.value) : '' })}
                    className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 font-mono focus:outline-none placeholder-slate-400"
                  />
                </div>
                
                {/* Atribuição de Técnicos com Atalho */}
                <div>
                  <div className="flex items-center justify-between mb-1.5">
                    <label htmlFor="formTech" className="block text-xs font-semibold text-slate-600 uppercase tracking-wide">
                      Técnico Atribuído
                    </label>
                    <button
                      type="button"
                      onClick={() => setIsTechModalOpen(true)}
                      className="text-[10px] text-blue-700 hover:text-blue-800 font-semibold"
                    >
                      + Novo Técnico
                    </button>
                  </div>
                  {/* Conforme Requisito 5.1: Seletor de Técnicos corrigido com correspondência exata */}
                  <select
                    id="formTech"
                    value={formData.assignedTo || ''}
                    onChange={(e) => setFormData({...formData, assignedTo: e.target.value})}
                    className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 cursor-pointer focus:outline-none"
                  >
                    <option value="">Selecione o Técnico / Mecânico...</option>
                    {technicians.map((t) => (
                      <option key={t.id} value={t.id}>{t.name} — {t.specialty}</option>
                    ))}
                  </select>
                </div>
              </div>

              {/* Secção de Peças Requisitadas no Modal da OS (Requisito 5.2) */}
              <div className="border-t border-slate-200 pt-4 space-y-3">
                <div className="flex items-center gap-1.5 text-blue-700">
                  <Package size={14} />
                  <span className="text-xs font-bold uppercase tracking-wider text-slate-800">
                    Peças & Consumíveis Requisitados (Armazém)
                  </span>
                </div>

                <div className="bg-slate-50 p-3.5 rounded-md border border-slate-200 space-y-3">
                  {/* Linha de Inserção Rápida */}
                  <div className="grid grid-cols-1 sm:grid-cols-12 gap-2 items-end">
                    <div className="sm:col-span-7">
                      <label className="block text-[10px] font-semibold text-slate-600 mb-1 uppercase">Selecione a Peça</label>
                      <select
                        value={selectedPartId}
                        onChange={(e) => setSelectedPartId(e.target.value)}
                        className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-2.5 py-2 text-xs text-slate-900 cursor-pointer focus:outline-none"
                      >
                        <option value="">Selecione a Peça do Armazém...</option>
                        {availableParts.map((p) => (
                          <option key={p.id} value={p.id} disabled={Number(p.current_stock) <= 0}>
                            {p.name} {p.part_number ? `(${p.part_number})` : ''} — Stock: {p.current_stock} {p.unit_of_measure || 'un'} — {Number(p.unit_cost).toLocaleString()} MZN
                          </option>
                        ))}
                      </select>
                    </div>
                    <div className="sm:col-span-2">
                      <label className="block text-[10px] font-semibold text-slate-600 mb-1 uppercase">Qtd</label>
                      <input
                        type="number"
                        min={1}
                        value={modalPartQty}
                        onChange={(e) => setModalPartQty(Number(e.target.value))}
                        className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-2 py-2 text-xs text-slate-900 font-mono focus:outline-none"
                      />
                    </div>
                    <div className="sm:col-span-3">
                      <button
                        type="button"
                        onClick={handleAddPartToModalList}
                        className="w-full py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium transition-colors cursor-pointer shadow-none"
                      >
                        + Adicionar
                      </button>
                    </div>
                  </div>

                  {/* Lista de Peças Adicionadas */}
                  {selectedPartsList.length === 0 ? (
                    <p className="text-[11px] text-slate-500 italic text-center py-2 bg-white rounded border border-slate-200">
                      Nenhuma peça ou consumível requisitado para esta intervenção.
                    </p>
                  ) : (
                    <div className="space-y-2">
                      <div className="max-h-[160px] overflow-y-auto space-y-1.5">
                        {selectedPartsList.map((p) => {
                          const subtotal = p.quantity * p.unit_cost;
                          return (
                            <div key={p.id} className="flex justify-between items-center text-[11px] bg-white p-2 rounded-md border border-slate-200">
                              <div className="truncate pr-2">
                                <span className="font-mono text-slate-500 text-[10px] block">[{p.part_number}]</span>
                                <span className="font-medium text-slate-900 block truncate">{p.name}</span>
                              </div>
                              <div className="flex items-center gap-3 shrink-0">
                                <div className="text-right">
                                  <span className="text-slate-700 font-semibold block">Qtd: {p.quantity}</span>
                                  <span className="text-blue-700 text-[10px] font-bold block">{formatMZN(subtotal)}</span>
                                </div>
                                <button
                                  type="button"
                                  onClick={() => setSelectedPartsList(selectedPartsList.filter(item => item.id !== p.id))}
                                  className="p-1 text-slate-400 hover:text-red-700 hover:bg-red-50 rounded transition-colors"
                                  title="Remover Peça"
                                >
                                  <X size={13} />
                                </button>
                              </div>
                            </div>
                          );
                        })}
                      </div>

                      <div className="pt-2 border-t border-slate-200 flex justify-between items-center text-[11px] font-bold text-slate-800">
                        <span>Subtotal das Peças Requisitadas:</span>
                        <span className="text-blue-700 font-bold">{formatMZN(selectedPartsList.reduce((acc, p) => acc + (p.quantity * p.unit_cost), 0))}</span>
                      </div>
                    </div>
                  )}
                </div>
              </div>

              <div className="flex justify-end gap-3 border-t border-slate-200 pt-5 mt-2">
                <button
                  type="button"
                  onClick={() => setIsFormModalOpen(false)}
                  disabled={submittingForm}
                  className="px-4 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 rounded-md text-xs font-medium transition-all cursor-pointer"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  disabled={submittingForm}
                  className="px-4 py-2 bg-blue-700 hover:bg-blue-800 active:bg-blue-900 text-white rounded-md text-xs font-medium transition-all flex items-center gap-1.5 cursor-pointer disabled:opacity-50 shadow-none"
                >
                  {submittingForm ? (
                    <Loader2 size={14} className="animate-spin" />
                  ) : (
                    <span>Guardar Ordem</span>
                  )}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL DE CADASTRO RÁPIDO: NOVO TÉCNICO
          ========================================== */}
      {isTechModalOpen && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-sm rounded-md shadow-xl overflow-hidden animate-slide-up">
            
            <div className="p-4 border-b border-slate-200 flex items-center justify-between bg-white">
              <div className="flex items-center gap-2">
                <UserCheck className="text-blue-700" size={18} />
                <span className="font-bold text-sm text-slate-900">Cadastrar Novo Técnico</span>
              </div>
              <button
                type="button"
                onClick={() => setIsTechModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors"
              >
                <X size={18} />
              </button>
            </div>

            <form onSubmit={handleCreateTechnicianSubmit} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-medium text-slate-700 mb-1">Nome Completo *</label>
                <input
                  type="text"
                  required
                  value={techName}
                  onChange={(e) => setTechName(e.target.value)}
                  placeholder="ex: Armando Sitoe"
                  className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                />
              </div>

              <div>
                <label className="block text-xs font-medium text-slate-700 mb-1">Especialidade / Ofício *</label>
                <select
                  required
                  value={techSpecialty}
                  onChange={(e) => setTechSpecialty(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600 cursor-pointer"
                >
                  <option value="">Selecione a Especialidade...</option>
                  {technicianSpecialties.map((spec) => (
                    <option key={spec.value} value={spec.value}>
                      {spec.label}
                    </option>
                  ))}
                </select>
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">Telefone</label>
                  <input
                    type="text"
                    value={techPhone}
                    onChange={(e) => setTechPhone(e.target.value)}
                    placeholder="ex: +258 84 123 4567"
                    className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 font-mono focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                  />
                </div>
                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">Custo / Hora (MZN)</label>
                  <input
                    type="number"
                    value={techCostPerHour}
                    onChange={(e) => setTechCostPerHour(e.target.value !== '' ? Number(e.target.value) : '')}
                    placeholder="ex: 1500"
                    className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 font-mono focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                  />
                </div>
              </div>

              <div className="flex justify-end gap-3 pt-3 border-t border-slate-200">
                <button
                  type="button"
                  onClick={() => setIsTechModalOpen(false)}
                  className="px-3.5 py-1.5 bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 text-xs font-medium rounded-md"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  disabled={submittingTech}
                  className="px-3.5 py-1.5 bg-blue-700 hover:bg-blue-800 text-white text-xs font-medium rounded-md flex items-center gap-1 cursor-pointer disabled:opacity-50"
                >
                  {submittingTech ? <Loader2 size={13} className="animate-spin" /> : <span>Guardar Técnico</span>}
                </button>
              </div>
            </form>

          </div>
        </div>
      )}

      {/* ==========================================
          DRAWER / BOTTOM SHEET: COCKPIT DE EXECUÇÃO DE CAMPO
          ========================================== */}
      {isDrawerOpen && selectedDrawerOrder && (
        <div className="fixed inset-0 z-40 flex justify-end bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div 
            className="absolute inset-0" 
            onClick={() => {
              setIsDrawerOpen(false);
              setSelectedDrawerOrder(null);
            }} 
          />
          
          <div className="relative w-full max-w-lg bg-white border-l border-slate-200 h-full shadow-2xl flex flex-col z-10 animate-slide-up text-slate-900">
            
            {/* Header do Drawer */}
            <div className="p-5 border-b border-slate-200 flex items-center justify-between bg-white shrink-0">
              <div className="space-y-1">
                <span className="font-mono font-bold text-xs text-blue-700 bg-blue-50 px-2 py-0.5 rounded-md border border-blue-200">
                  {selectedDrawerOrder.wo_number || selectedDrawerOrder.woNumber || 'ORDEM'}
                </span>
                <h2 className="text-base font-bold text-slate-900 leading-tight mt-1">{selectedDrawerOrder.title}</h2>
              </div>
              <button
                onClick={() => {
                  setIsDrawerOpen(false);
                  setSelectedDrawerOrder(null);
                }}
                className="p-1.5 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors"
              >
                <X size={20} />
              </button>
            </div>

            {/* Corpo do Drawer (Rolagem) */}
            <div className="flex-1 overflow-y-auto p-5 space-y-5">
              
              {/* Informações Básicas */}
              <div className="p-4 bg-slate-50 border border-slate-200 rounded-md space-y-3">
                <div className="grid grid-cols-2 gap-3 text-xs">
                  <div>
                    <span className="text-slate-500 uppercase text-[10px] font-semibold block">Ativo Vinculado</span>
                    <span className="text-slate-900 font-mono font-bold block truncate">
                      {assets.find(a => a.id === selectedDrawerOrder.asset_id || a.id === selectedDrawerOrder.assetId)?.name || selectedDrawerOrder.asset_name || 'Equipamento Geral'}
                    </span>
                  </div>
                  <div>
                    <span className="text-slate-500 uppercase text-[10px] font-semibold block">Urgência da Manutenção</span>
                    <div className="mt-0.5">{renderPriorityBadge(selectedDrawerOrder.priority)}</div>
                  </div>
                  <div>
                    <span className="text-slate-500 uppercase text-[10px] font-semibold block">Técnico Atribuído</span>
                    <span className="text-slate-900 font-semibold">
                      {resolveTechnicianName(selectedDrawerOrder)}
                    </span>
                  </div>
                  <div>
                    <span className="text-slate-500 uppercase text-[10px] font-semibold block">Estado Técnico Atual</span>
                    <div className="mt-0.5">{renderStatusBadge(selectedDrawerOrder.status)}</div>
                  </div>
                </div>

                {selectedDrawerOrder.description && (
                  <div className="pt-2.5 border-t border-slate-200">
                    <span className="text-slate-500 uppercase text-[10px] font-semibold block mb-1">Descrição Técnico-Operacional</span>
                    <p className="text-slate-700 text-xs leading-relaxed whitespace-pre-line">{selectedDrawerOrder.description}</p>
                  </div>
                )}
              </div>

              {/* COCKPIT DE EXECUÇÃO DE CAMPO EM TEMPO REAL */}
              {selectedDrawerOrder.status !== 'completed' && (
                <div className="p-4 bg-white border border-slate-200 rounded-md space-y-3">
                  <span className="text-xs font-bold text-slate-900 uppercase tracking-wide block flex items-center gap-1.5">
                    <Activity size={14} className="text-blue-700" />
                    <span>Cockpit de Execução de Campo</span>
                  </span>

                  <div className="grid grid-cols-1 gap-2.5">
                    {/* SE STATUS = 'open' */}
                    {selectedDrawerOrder.status === 'open' && (
                      <button
                        onClick={() => handleStartService(selectedDrawerOrder)}
                        className="w-full py-2.5 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-2 transition-colors cursor-pointer shadow-none"
                      >
                        <Play size={15} />
                        <span>Iniciar Serviço</span>
                      </button>
                    )}

                    {/* SE STATUS = 'in_progress' */}
                    {selectedDrawerOrder.status === 'in_progress' && (
                      <div className="grid grid-cols-2 gap-2.5">
                        <button
                          type="button"
                          onClick={() => handleOpenPauseModal(selectedDrawerOrder)}
                          className="py-2.5 bg-amber-600 hover:bg-amber-700 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-colors cursor-pointer"
                        >
                          <Pause size={15} />
                          <span>Pausar Trabalho</span>
                        </button>
                        <button
                          type="button"
                          onClick={() => {
                            setSelectedOrder(selectedDrawerOrder);
                            
                            // Calcula e preenche valores automaticamente
                            const materialsCost = usedParts.reduce((acc, p) => acc + (p.subtotal || 0), 0);
                            const tech = technicians.find(t => t.id === selectedDrawerOrder.technician_id || t.id === selectedDrawerOrder.technicianId);
                            const hourlyRate = tech?.cost_per_hour || tech?.costPerHour || 0;
                            const workedHours = calculateWorkedHours();
                            const laborCost = workedHours * hourlyRate;

                            setPartsCostMzn(materialsCost > 0 ? materialsCost : '');
                            setLaborCostMzn(laborCost > 0 ? laborCost : '');
                            setDowntimeHours(workedHours > 0 ? workedHours : '');
                            
                            setIsCloseModalOpen(true);
                          }}
                          className="py-2.5 bg-emerald-700 hover:bg-emerald-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-colors cursor-pointer"
                        >
                          <CheckCircle2 size={15} />
                          <span>Concluir Serviço</span>
                        </button>
                      </div>
                    )}

                    {/* SE STATUS = 'waiting_parts' ou 'paused' ou 'blocked' */}
                    {(selectedDrawerOrder.status === 'waiting_parts' || selectedDrawerOrder.status === 'paused' || selectedDrawerOrder.status === 'blocked') && (
                      <button
                        onClick={() => handleResumeService(selectedDrawerOrder)}
                        className="w-full py-2.5 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-2 transition-colors cursor-pointer"
                      >
                        <Play size={15} />
                        <span>Retomar Trabalho (Resume)</span>
                      </button>
                    )}
                  </div>
                </div>
              )}

              {/* PEÇAS E CONSUMÍVEIS UTILIZADOS (REQUISITO 10.2) */}
              <div className="p-4 bg-white border border-slate-200 rounded-md space-y-4">
                <div className="flex items-center justify-between border-b border-slate-200 pb-2">
                  <span className="text-xs font-bold text-slate-700 uppercase tracking-wide block">
                    Peças & Consumíveis Utilizados ({usedParts.length})
                  </span>
                  
                  {/* Se a OS estiver em execução ('in_progress') e o utilizador for o técnico ou supervisor */}
                  {selectedDrawerOrder.status === 'in_progress' && (
                    <button
                      type="button"
                      onClick={() => {
                        setSelectedPartId('');
                        setPartQtyToUse('');
                        setIsAddPartModalOpen(true);
                      }}
                      className="text-xs font-medium text-blue-700 hover:text-blue-800 flex items-center gap-1 cursor-pointer"
                    >
                      <Plus size={13} />
                      <span>Adicionar Peça</span>
                    </button>
                  )}
                </div>

                {loadingUsedParts ? (
                  <div className="text-center py-4">
                    <Loader2 size={16} className="animate-spin text-slate-400 mx-auto" />
                  </div>
                ) : usedParts.length === 0 ? (
                  <div className="text-center py-3">
                    <p className="text-xs text-slate-500 italic">Nenhum consumível apontado a esta intervenção ainda.</p>
                  </div>
                ) : (
                  <div className="space-y-2.5">
                    {usedParts.map((item) => (
                      <div key={item.id} className="bg-slate-50 border border-slate-200 p-2.5 rounded-md flex items-center justify-between text-xs hover:border-slate-300 transition-colors">
                        <div className="space-y-0.5 truncate mr-3">
                          <span className="font-semibold text-slate-900 block truncate">{item.name}</span>
                          <span className="text-[11px] text-slate-500 font-mono block">
                            Ref: {item.oem_reference} · Quantidade: <span className="font-bold text-slate-700">{item.quantity} un</span>
                          </span>
                        </div>
                        <div className="text-right shrink-0">
                          <span className="text-[10px] text-slate-500 block">Subtotal:</span>
                          <span className="font-mono text-blue-700 font-bold block">{formatMZN(item.subtotal)}</span>
                        </div>
                      </div>
                    ))}
                    
                    {/* Exibe totalizador de peças */}
                    <div className="pt-2 border-t border-slate-200 flex justify-between items-center text-xs font-bold text-slate-700">
                      <span>Custo Acumulado de Peças:</span>
                      <span className="font-mono text-slate-900 font-bold">
                        {formatMZN(usedParts.reduce((acc, p) => acc + (p.subtotal || 0), 0))}
                      </span>
                    </div>
                  </div>
                )}
              </div>

              {/* TOTALIZADORES DE CUSTO DA INTERVENÇÃO (REQUISITO 10.3) */}
              <div className="p-4 bg-white border border-slate-200 rounded-md space-y-3">
                <span className="text-xs font-bold text-slate-700 uppercase tracking-wide block border-b border-slate-200 pb-2 flex items-center gap-1.5">
                  <DollarSign size={14} className="text-blue-700" />
                  <span>Totalizadores de Custo da Intervenção</span>
                </span>
                
                {(() => {
                  const materialsCost = usedParts.reduce((acc, p) => acc + (p.subtotal || 0), 0);
                  const tech = technicians.find(t => t.id === selectedDrawerOrder.technician_id || t.id === selectedDrawerOrder.technicianId);
                  const hourlyRate = tech?.cost_per_hour || tech?.costPerHour || 0;
                  const workedHours = calculateWorkedHours();
                  const laborCost = workedHours * hourlyRate;
                  const totalCostOfOS = materialsCost + laborCost;

                  return (
                    <div className="space-y-2 text-xs">
                      <div className="flex justify-between items-center text-slate-600">
                        <span>Custo de Materiais (Peças):</span>
                        <span className="font-mono text-slate-900 font-semibold">{formatMZN(materialsCost)}</span>
                      </div>
                      <div className="flex justify-between items-center text-slate-600">
                        <div className="space-y-0.5">
                          <span>Mão de Obra do Técnico:</span>
                          {tech && (
                            <span className="text-[11px] text-slate-500 block">
                              {tech.name} ({formatMZN(hourlyRate)}/h)
                            </span>
                          )}
                        </div>
                        <div className="text-right">
                          <span className="font-mono text-slate-900 font-semibold">{formatMZN(laborCost)}</span>
                          <span className="text-[10px] text-slate-500 block font-mono">({workedHours} horas)</span>
                        </div>
                      </div>
                      <div className="pt-2 border-t border-slate-200 flex justify-between items-center text-sm font-bold text-slate-900">
                        <span className="text-slate-900">Custo Total da OS:</span>
                        <span className="font-mono text-blue-700 bg-blue-50 border border-blue-200 px-2 py-0.5 rounded-md">
                          {formatMZN(totalCostOfOS)}
                        </span>
                      </div>
                    </div>
                  );
                })()}
              </div>

              {/* LINHA DO TEMPO DE AUDITORIA */}
              <div className="p-4 bg-white border border-slate-200 rounded-md space-y-4">
                <span className="text-xs font-bold text-slate-700 uppercase tracking-wide block border-b border-slate-200 pb-2">
                  Linha do Tempo de Auditoria (Timeline)
                </span>

                {loadingTimeline ? (
                  <div className="text-center py-4">
                    <Loader2 size={18} className="animate-spin text-slate-400 mx-auto" />
                  </div>
                ) : timeline.length === 0 ? (
                  <div className="text-center py-3">
                    <p className="text-xs text-slate-500 italic">Nenhum evento registrado nesta OS ainda.</p>
                  </div>
                ) : (
                  <div className="relative pl-6 border-l-2 border-slate-200 space-y-4">
                    {timeline.map((event) => {
                      const date = event.created_at || event.createdAt;
                      const type = event.eventType || event.event_type;
                      
                      let iconColor = 'bg-slate-100 text-slate-600 border border-slate-200';
                      if (type === 'start' || type === 'resume') iconColor = 'bg-blue-50 text-blue-700 border border-blue-200';
                      if (type === 'pause') iconColor = 'bg-amber-50 text-amber-700 border border-amber-200';
                      if (type === 'complete') iconColor = 'bg-emerald-50 text-emerald-700 border border-emerald-200';

                      return (
                        <div key={event.id} className="relative">
                          {/* Dot / Icon */}
                          <div className={`absolute -left-[35px] top-0 w-6 h-6 rounded-full flex items-center justify-center text-[10px] ${iconColor}`}>
                            {type === 'start' || type === 'resume' ? <Play size={10} /> : type === 'pause' ? <Pause size={10} /> : <CheckCircle2 size={10} />}
                          </div>

                          <div className="space-y-0.5 text-xs">
                            <div className="flex justify-between items-center">
                              <span className="font-bold text-slate-900 capitalize">
                                {type === 'start' ? 'Serviço Iniciado' : type === 'pause' ? 'Serviço Pausado' : type === 'resume' ? 'Serviço Retomado' : 'Serviço Concluído'}
                              </span>
                              <span className="text-[10px] text-slate-500 font-mono">
                                {date ? new Date(date).toLocaleString('pt-MZ') : '---'}
                              </span>
                            </div>
                            <div className="text-xs text-slate-600">
                              <span>Executado por: </span>
                              <span className="font-semibold text-slate-900">{event.performed_by || event.performedBy || 'Técnico de Campo'}</span>
                            </div>
                            {event.reason && (
                              <div className="text-xs font-medium text-amber-700 mt-0.5">
                                Motivo: {event.reason}
                              </div>
                            )}
                            {event.justification && (
                              <p className="text-xs text-slate-600 leading-relaxed italic bg-slate-50 p-2 rounded-md border border-slate-200 mt-1">
                                "{event.justification}"
                              </p>
                            )}
                          </div>
                        </div>
                      );
                    })}
                  </div>
                )}
              </div>

            </div>

          </div>
        </div>
      )}

      {/* ==========================================
          MODAL OBRIGATÓRIO: PAUSA / INTERRUPÇÃO
          ========================================== */}
      {isPauseModalOpen && selectedDrawerOrder && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-sm rounded-md shadow-xl overflow-hidden animate-slide-up">
            
            <div className="p-4 border-b border-slate-200 flex items-center justify-between bg-white">
              <div className="flex items-center gap-2 text-amber-700">
                <Pause size={18} />
                <span className="font-bold text-sm text-slate-900">Justificar Pausa de Serviço</span>
              </div>
              <button
                type="button"
                onClick={() => setIsPauseModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors"
              >
                <X size={18} />
              </button>
            </div>

            <form onSubmit={handlePauseServiceSubmit} className="p-5 space-y-4">
              <div>
                <label htmlFor="pauseSelect" className="block text-xs font-medium text-slate-700 mb-1">Motivo Padronizado *</label>
                <select
                  id="pauseSelect"
                  value={pauseReason}
                  onChange={(e) => setPauseReason(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 cursor-pointer focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                >
                  <option value="Aguardando Peças no Armazém">Aguardando Peças no Armazém</option>
                  <option value="Corte de Energia / Gerador">Corte de Energia / Gerador</option>
                  <option value="Fim de Turno">Fim de Turno</option>
                  <option value="Prioridade Operacional / Emergência">Prioridade Operacional / Emergência</option>
                  <option value="Falta de Ferramenta Especial">Falta de Ferramenta Especial</option>
                  <option value="Outro">Outro</option>
                </select>
              </div>

              <div>
                <label htmlFor="pauseJust" className="block text-xs font-medium text-slate-700 mb-1">Justificação Técnica Detalhada *</label>
                <textarea
                  id="pauseJust"
                  required
                  rows={3}
                  value={pauseJustification}
                  onChange={(e) => setPauseJustification(e.target.value)}
                  placeholder="Por favor, explique circunstancialmente o motivo da interrupção do trabalho..."
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md p-2.5 text-xs text-slate-900 focus:outline-none"
                />
              </div>

              <div className="flex justify-end gap-3 pt-3 border-t border-slate-200">
                <button
                  type="button"
                  onClick={() => setIsPauseModalOpen(false)}
                  className="px-3.5 py-1.5 bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 text-xs font-medium rounded-md"
                >
                  Voltar
                </button>
                <button
                  type="submit"
                  disabled={submittingPause || !pauseJustification.trim()}
                  className="px-3.5 py-1.5 bg-amber-600 hover:bg-amber-700 text-white text-xs font-medium rounded-md flex items-center gap-1 cursor-pointer disabled:opacity-50"
                >
                  {submittingPause ? <Loader2 size={13} className="animate-spin" /> : <span>Confirmar Pausa</span>}
                </button>
              </div>
            </form>

          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: CONCLUIR / FECHAR ORDEM DE SERVIÇO
          ========================================== */}
      {isCloseModalOpen && selectedOrder && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-md rounded-md shadow-xl overflow-hidden animate-slide-up">
            
            <div className="p-5 border-b border-slate-200 flex items-center justify-between bg-white">
              <div className="flex items-center gap-2 text-emerald-700">
                <CheckCircle2 size={18} />
                <h2 className="text-base font-bold text-slate-900">Concluir Ordem de Serviço</h2>
              </div>
              <button
                type="button"
                onClick={() => {
                  setIsCloseModalOpen(false);
                  setSelectedOrder(null);
                }}
                className="p-1.5 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors"
              >
                <X size={18} />
              </button>
            </div>

            <form onSubmit={handleCloseOrderSubmit} className="p-6 space-y-4 max-h-[75vh] overflow-y-auto">
              
              <div className="bg-slate-50 border border-slate-200 p-3 rounded-md text-xs space-y-1.5 text-slate-600">
                <div><span className="font-semibold text-slate-700">Nº da OS:</span> <span className="font-mono text-slate-900 font-bold">{selectedOrder.wo_number || selectedOrder.woNumber}</span></div>
                <div><span className="font-semibold text-slate-700">Atividade:</span> <span className="text-slate-900">{selectedOrder.title}</span></div>
              </div>

              <div>
                <label htmlFor="notes" className="block text-xs font-medium text-slate-700 mb-1.5">
                  Notas de Resolução / Trabalho Executado *
                </label>
                <textarea
                  id="notes"
                  required
                  rows={2}
                  value={resolutionNotes}
                  onChange={(e) => setResolutionNotes(e.target.value)}
                  placeholder="Descreva detalhadamente o trabalho executado e as peças substituídas..."
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                />
              </div>

              <div>
                <label htmlFor="downtime" className="block text-xs font-medium text-slate-700 mb-1.5">
                  Downtime / Horas de Paragem do Ativo *
                </label>
                <div className="relative">
                  <input
                    id="downtime"
                    type="number"
                    step="0.1"
                    min={0}
                    required
                    value={downtimeHours}
                    onChange={(e) => setDowntimeHours(e.target.value !== '' ? Number(e.target.value) : '')}
                    placeholder="ex: 4.5"
                    className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3.5 py-2 text-xs text-slate-900 font-mono focus:outline-none"
                  />
                  <span className="absolute right-3.5 top-2.5 text-[10px] font-bold text-slate-500">HORAS</span>
                </div>
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label htmlFor="partsCost" className="block text-xs font-medium text-slate-700 mb-1.5">
                    Custo de Peças (MZN)
                  </label>
                  <div className="relative">
                    <input
                      id="partsCost"
                      type="number"
                      min={0}
                      value={partsCostMzn}
                      onChange={(e) => setPartsCostMzn(e.target.value !== '' ? Number(e.target.value) : '')}
                      placeholder="ex: 15000"
                      className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 font-mono focus:outline-none"
                    />
                    <span className="absolute right-3.5 top-2.5 text-[10px] font-bold text-slate-500">MZN</span>
                  </div>
                </div>
                <div>
                  <label htmlFor="laborCost" className="block text-xs font-medium text-slate-700 mb-1.5">
                    Custo de Mão de Obra (MZN)
                  </label>
                  <div className="relative">
                    <input
                      id="laborCost"
                      type="number"
                      min={0}
                      value={laborCostMzn}
                      onChange={(e) => setLaborCostMzn(e.target.value !== '' ? Number(e.target.value) : '')}
                      placeholder="ex: 4500"
                      className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 font-mono focus:outline-none"
                    />
                    <span className="absolute right-3.5 top-2.5 text-[10px] font-bold text-slate-500">MZN</span>
                  </div>
                </div>
              </div>

              <div className="flex justify-end gap-3 border-t border-slate-200 pt-4 mt-2">
                <button
                  type="button"
                  onClick={() => {
                    setIsCloseModalOpen(false);
                    setSelectedOrder(null);
                  }}
                  disabled={submittingClose}
                  className="px-4 py-2 bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 rounded-md text-xs font-medium transition-colors cursor-pointer"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  disabled={submittingClose}
                  className="px-4 py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium transition-colors flex items-center gap-1.5 cursor-pointer disabled:opacity-50"
                >
                  {submittingClose ? (
                    <Loader2 size={14} className="animate-spin" />
                  ) : (
                    <span>Concluir Ordem</span>
                  )}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: APONTAR PEÇA UTILIZADA (REQUISITO 10.2)
          ========================================== */}
      {isAddPartModalOpen && selectedDrawerOrder && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-sm rounded-md shadow-xl overflow-hidden animate-slide-up">
            
            <div className="p-4 border-b border-slate-200 flex items-center justify-between bg-white">
              <div className="flex items-center gap-2">
                <Package className="text-blue-700" size={18} />
                <span className="font-bold text-sm text-slate-900">Apontar Peça Utilizada</span>
              </div>
              <button
                type="button"
                onClick={() => setIsAddPartModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors"
              >
                <X size={18} />
              </button>
            </div>

            <form onSubmit={handleAddPartSubmit} className="p-5 space-y-4">
              <div>
                <label htmlFor="selectPart" className="block text-xs font-medium text-slate-700 mb-1">Peça em Stock *</label>
                <select
                  id="selectPart"
                  required
                  value={selectedPartId}
                  onChange={(e) => setSelectedPartId(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded-md px-2.5 py-1.5 text-xs text-slate-900 cursor-pointer focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                >
                  <option value="">Selecione o Artigo...</option>
                  {inventoryParts.map((p) => {
                    const stk = p.current_stock ?? p.currentStock ?? 0;
                    return (
                      <option key={p.id} value={p.id} disabled={stk === 0}>
                        {p.name} [{p.oem_reference || p.oemReference}] (Stock: {stk} un)
                      </option>
                    );
                  })}
                </select>
              </div>

              {selectedPartId && (
                <div className="bg-slate-50 border border-slate-200 p-3 rounded-md text-xs space-y-1 text-slate-600">
                  {(() => {
                    const p = inventoryParts.find(item => item.id === selectedPartId);
                    if (!p) return null;
                    return (
                      <>
                        <div><span className="font-semibold text-slate-700">OEM / Ref:</span> <span className="font-mono text-blue-700">{p.oem_reference || p.oemReference}</span></div>
                        <div><span className="font-semibold text-slate-700">Custo Unitário:</span> <span className="text-slate-900 font-bold">{formatMZN(p.unit_cost ?? p.unitCost)}</span></div>
                        <div><span className="font-semibold text-slate-700">Localização:</span> <span className="text-slate-700">{p.warehouse_location || p.warehouseLocation || 'Geral'}</span></div>
                      </>
                    );
                  })()}
                </div>
              )}

              <div>
                <label htmlFor="qtyToUse" className="block text-xs font-medium text-slate-700 mb-1">Quantidade Requerida *</label>
                <input
                  id="qtyToUse"
                  type="number"
                  required
                  min={1}
                  value={partQtyToUse}
                  onChange={(e) => setPartQtyToUse(e.target.value !== '' ? Number(e.target.value) : '')}
                  placeholder="unidades a retirar do stock"
                  className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-1.5 text-xs text-slate-900 focus:outline-none"
                />
              </div>

              <div className="flex justify-end gap-3 pt-3 border-t border-slate-200">
                <button
                  type="button"
                  onClick={() => setIsAddPartModalOpen(false)}
                  className="px-3.5 py-1.5 bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 text-xs font-medium rounded-md"
                >
                  Voltar
                </button>
                <button
                  type="submit"
                  disabled={submittingAddPart || !selectedPartId || partQtyToUse === ''}
                  className="px-3.5 py-1.5 bg-blue-700 hover:bg-blue-800 text-white text-xs font-medium rounded-md flex items-center gap-1 cursor-pointer disabled:opacity-50"
                >
                  {submittingAddPart ? <Loader2 size={13} className="animate-spin" /> : <span>Confirmar Apontamento</span>}

      let hasEmptyName = false;

        const currentLine = lines[i].trim();
        if (!currentLine) continue;

        const values = currentLine.split(delimiter).map((v) => v.trim().replace(/^"|"$/g, ''));
        const rowObj: Record<string, string> = {};

        headers.forEach((header, index) => {
          rowObj[header] = values[index] || '';
        });

        const nomeValue = rowObj['Nome'] || rowObj['nome'] || rowObj['NOME'];
        if (!nomeValue || !nomeValue.trim()) {
          hasEmptyName = true;
        }

        rows.push(rowObj);
      }

      if (hasEmptyName) {
        setBatchError('Validação de Dados: Foram encontradas linhas sem o campo "Nome" preenchido. O campo Nome é obrigatório para todos os ativos.');
      }

      setParsedBatchData(rows);
    };

    reader.onerror = () => {
      setBatchError('Falha ao efetuar a leitura do ficheiro CSV.');
    };

    reader.readAsText(file);
  };

  const handleBatchFileSelect = (file: File) => {
    if (!file.name.toLowerCase().endsWith('.csv') && file.type !== 'text/csv' && file.type !== 'application/vnd.ms-excel') {
      setBatchError('Formato inválido. Por favor, carregue um ficheiro no formato CSV (.csv).');
      return;
    }
    setBatchFile(file);
    parseCsvFile(file);
  };

  const handleBatchDragOver = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setBatchDragActive(true);
  };

  const handleBatchDragLeave = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setBatchDragActive(false);
  };

  const handleBatchDrop = (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setBatchDragActive(false);
    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
      handleBatchFileSelect(e.dataTransfer.files[0]);
    }
  };

  const handleBatchSubmit = async () => {
    if (parsedBatchData.length === 0) {
      setBatchError('Nenhum registo válido encontrado no ficheiro CSV para importar.');
      return;
    }

    const invalidIndex = parsedBatchData.findIndex(
      row => !(row.Nome || row.nome || row.NOME)?.trim()
    );

    if (invalidIndex !== -1) {
      setBatchError(`A linha ${invalidIndex + 1} possui o campo "Nome" em branco. Corrija o ficheiro antes de importar.`);
      return;
    }

    setBatchSubmitting(true);
    setBatchError(null);

    const assetsToImport = parsedBatchData.map((row) => {
      const isVehicle = currentScope === 'fleet' || row.Categoria === 'vehicle_light' || row.Categoria === 'vehicle_heavy' || row.categoria === 'vehicle_light' || row.categoria === 'vehicle_heavy';
      const cat = row.Categoria || row.categoria || (isVehicle ? 'vehicle_light' : 'generator');
      const meterKind = isVehicle ? 'odometer_km' : 'hourmeter_hours';

      return {
        name: (row.Nome || row.nome || '').trim(),
        category: cat,
        codeTag: (row.Tag || row.tag || row.codeTag || row.code_tag || '').trim() || undefined,
        code_tag: (row.Tag || row.tag || row.codeTag || row.code_tag || '').trim() || undefined,
        brand: (row.Marca || row.marca || row.Fabricante || row.fabricante || '').trim() || undefined,
        model: (row.Modelo || row.modelo || '').trim() || '',
        manufactureYear: row.Ano ? Number(row.Ano) : new Date().getFullYear(),
        manufacture_year: row.Ano ? Number(row.Ano) : new Date().getFullYear(),
        costCenter: 'Sede / Geral',
        cost_center: 'Sede / Geral',
        meterKind,
        meter_kind: meterKind,
        currentMeter: Number(row.OdometroInicial || row.HorimetroInicial || row.odometroinicial || row.horimetroinicial || row.currentMeter || row.current_meter || 0),
        current_meter: Number(row.OdometroInicial || row.HorimetroInicial || row.odometroinicial || row.horimetroinicial || row.currentMeter || row.current_meter || 0),
        licensePlate: (row.Matricula || row.matricula || '').trim() || undefined,
        license_plate: (row.Matricula || row.matricula || '').trim() || undefined,
        chassisVin: (row.Chassis || row.chassis || '').trim() || undefined,
        chassis_vin: (row.Chassis || row.chassis || '').trim() || undefined,
        plantSector: (row.SetorFabril || row.setorfabril || '').trim() || undefined,
        plant_sector: (row.SetorFabril || row.setorfabril || '').trim() || undefined,
        powerKwKva: row.Potencia || row.potencia ? Number(row.Potencia || row.potencia) : undefined,
        power_kw_kva: row.Potencia || row.potencia ? Number(row.Potencia || row.potencia) : undefined,
        voltageSpecs: (row.Voltagem || row.voltagem || '').trim() || undefined,
        voltage_specs: (row.Voltagem || row.voltagem || '').trim() || undefined,
        status: 'operational'
      };
    });

    try {
      let response;
      try {
        response = await api.post('/assets/batch', { assets: assetsToImport });
      } catch (firstErr: any) {
        if (firstErr.response?.status === 404) {
          try {
            response = await api.post('/assets/import', { assets: assetsToImport });
          } catch (secondErr: any) {
            if (secondErr.response?.status === 404) {
              for (const assetPayload of assetsToImport) {
                await api.post('/assets', assetPayload);
              }
              response = { data: { success: true } };
            } else {
              throw secondErr;
            }
          }
        } else {
          throw firstErr;
        }
      }

      const count = assetsToImport.length;
      setSuccessMessage(`${count} ${count === 1 ? 'ativo importado' : 'ativos importados'} em lote com sucesso!`);
      
      setIsBatchImportModalOpen(false);
      setBatchFile(null);
      setParsedBatchData([]);
      setBatchError(null);

      await fetchAssets();
      setTimeout(() => setSuccessMessage(null), 4000);
    } catch (err: any) {
      console.error('Erro ao importar ativos em lote:', err);
      const errorDetail = err.response?.data?.message || err.response?.data?.error || err.message || 'Falha ao processar a importação em lote dos ativos na API.';
      setBatchError(errorDetail);
    } finally {
      setBatchSubmitting(false);
    }
  };

  // ==========================================
  // AUXILIARES DE RENDERIZAÇÃO
  // ==========================================
  const formatCategory = (cat: string) => {
    switch (cat) {
      case 'vehicle_light':
        return 'Veículo Ligeiro';
      case 'vehicle_heavy':
        return 'Frota Pesada';
      case 'generator':
        return 'Gerador Industrial';
      case 'machinery':
        return 'Maquinaria / Chão Fábrica';
      default:
        return cat || 'Não especificado';
    }
  };

  const formatDocType = (type: string) => {
    switch (type) {
      case 'tech_sheet': return 'Ficha Técnica';
      case 'manual': return 'Manual Técnico';
      case 'diagram': return 'Diagrama Elétrico / Mecânico';
      case 'inspection_certificate': return 'Certificado de Inspeção (INATRO/INNOQ)';
      case 'insurance': return 'Apólice de Seguro';
      case 'asset_photo': return 'Fotografia do Ativo';
      default: return type;
    }
  };

  const renderStatusBadge = (status: string) => {
    switch (status) {
      case 'operational':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-md text-xs font-medium bg-emerald-50 text-emerald-700 border border-emerald-200">
            Disponível
          </span>
        );
      case 'breakdown':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-md text-xs font-medium bg-red-50 text-red-700 border border-red-200">
            Avariado
          </span>
        );
      case 'maintenance':
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-md text-xs font-medium bg-amber-50 text-amber-700 border border-amber-200">
            Em Oficina
          </span>
        );
      default:
        return (
          <span className="inline-flex items-center gap-1.5 px-2.5 py-0.5 rounded-md text-xs font-medium bg-slate-100 text-slate-700 border border-slate-200">
            {status}
          </span>
        );
    }
  };

  // Filtragem dos ativos localmente
  const filteredAssets = assets.filter((asset) => {
    const code = (asset.code_tag || asset.codeTag || '').toLowerCase();
    const assetName = (asset.name || '').toLowerCase();
    const assetModel = (asset.model || '').toLowerCase();
    const query = searchQuery.toLowerCase();

    const matchesSearch = code.includes(query) || assetName.includes(query) || assetModel.includes(query);
    const matchesCategory = categoryFilter === 'all' || asset.category === categoryFilter;

    // Filtro baseado no escopo ativo ('fleet' ou 'plant')
    const isAssetPlantType = asset.category === 'generator' || asset.category === 'machinery';
    const isAssetFleetType = asset.category === 'vehicle_light' || asset.category === 'vehicle_heavy';
    
    let matchesScope = true;
    if (currentScope === 'plant') {
      matchesScope = isAssetPlantType;
    } else if (currentScope === 'fleet') {
      matchesScope = isAssetFleetType;
    }

    return matchesSearch && matchesCategory && matchesScope;
  });

  // Se for driver_operator, redireciona para a interface simplificada tátil
  if (isDriverOperator) {
    return (
      <DriverOperatorPortal assets={assets} fetchAssets={fetchAssets} loading={loading} />
    );
  }

  return (
    <div className="space-y-6 print:hidden">
      
      {/* Toast de Sucesso */}
      {successMessage && (
        <div className="fixed bottom-6 right-6 z-50 bg-white border border-slate-200 text-slate-900 p-4 rounded-md shadow-lg max-w-md flex gap-3 items-start animate-fade-in">
          <CheckCircle2 className="text-emerald-700 shrink-0 mt-0.5" size={18} />
          <div>
            <span className="font-bold block text-sm text-slate-900">Sucesso</span>
            <span className="text-xs text-slate-600">{successMessage}</span>
          </div>
        </div>
      )}

      {/* Barra de Ações */}
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
        {loading ? (
          <div className="flex items-center gap-2 text-xs text-slate-500">
            <Loader2 size={16} className="animate-spin text-slate-400" />
            <span>A carregar...</span>
          </div>
        ) : <div />}
        <div className="flex gap-2">
          <button
            onClick={fetchAssets}
            disabled={loading}
            className="p-2.5 bg-white hover:bg-slate-50 border border-slate-300 text-slate-700 rounded-md transition-colors cursor-pointer"
          >
            <RefreshCw size={16} className={loading ? "animate-spin" : ""} />
          </button>
          {!isTechnician && !isDriverOperator && (
            <>
              <button
                onClick={() => {
                  setBatchFile(null);
                  setParsedBatchData([]);
                  setBatchError(null);
                  setIsBatchImportModalOpen(true);
                }}
                className="bg-white hover:bg-slate-50 border border-slate-300 text-slate-700 text-xs font-medium py-2.5 px-4 rounded-md flex items-center gap-1.5 transition-colors cursor-pointer shadow-none"
              >
                <FileSpreadsheet size={16} className="text-slate-500" />
                <span>Importar em Lote</span>
              </button>
              <button
                onClick={() => {
                  resetWizardForm();
                  setModalOpenMode('create');
                  setIsModalOpen(true);
                }}
                className="bg-blue-700 hover:bg-blue-800 text-white text-xs font-medium py-2.5 px-4 rounded-md flex items-center gap-1.5 transition-colors cursor-pointer shadow-none"
              >
                <Plus size={16} />
                <span>Cadastrar Ativo</span>
              </button>
            </>
          )}
        </div>
      </div>

      {/* Tratamento de erro */}
      {error && (
        <div className="p-4 bg-red-50 border border-red-200 text-red-700 text-xs rounded-md flex gap-3 items-start">
          <AlertTriangle size={18} className="text-red-600 shrink-0 mt-0.5" />
          <div className="flex-1">
            <span className="font-bold block text-red-800 mb-0.5">Erro na API</span>
            <span className="leading-relaxed block text-red-700">{error}</span>
          </div>
        </div>
      )}

      {/* Caixa de Busca e Filtros */}
      <div className="bg-white border border-slate-200 rounded-md p-4 flex flex-col sm:flex-row gap-3 shadow-none">
        <div className="relative flex-1">
          <Search size={14} className="absolute left-3.5 top-3 text-slate-400" />
          <input
            type="text"
            placeholder={searchPlaceholder}
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md pl-9 pr-3 py-2 text-xs text-slate-900 placeholder-slate-400 focus:outline-none"
          />
        </div>
        <div className="flex gap-2">
          <select
            value={categoryFilter}
            onChange={(e) => setCategoryFilter(e.target.value)}
            className="bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-700 focus:outline-none cursor-pointer"
          >
            <option value="all">Todas as Categorias</option>
            {categories.map((c) => (
              <option key={c.value} value={c.value}>
                {c.label}
              </option>
            ))}
          </select>
        </div>
      </div>

      {/* Tabela de Ativos com Ações Completas */}
      <div className="bg-white border border-slate-200 rounded-md overflow-hidden shadow-none">
        {loading ? (
          <div className="p-8 space-y-4">
            <div className="h-4 bg-slate-100 rounded w-1/4 animate-pulse"></div>
            {[1, 2, 3].map((n) => (
              <div key={n} className="flex gap-4 items-center py-2">
                <div className="h-8 bg-slate-100 rounded flex-1 animate-pulse"></div>
              </div>
            ))}
          </div>
        ) : filteredAssets.length === 0 ? (
          <div className="text-center py-16 px-4">
            <h3 className="text-sm font-semibold text-slate-900">Nenhum Ativo Encontrado</h3>
            <p className="text-slate-500 text-xs mt-1">
              {(user?.operational_scope || user?.tenant?.operational_scope || domain) === 'plant' 
                ? 'Registe equipamentos fabris para iniciar.' 
                : 'Registe viaturas e frotas para iniciar.'}
            </p>
          </div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-left border-collapse text-sm">
              <thead>
                <tr className="border-b border-slate-200 text-slate-600 text-xs uppercase bg-slate-50">
                  <th className="py-3 px-4 font-semibold">Código / Tag</th>
                  <th className="py-3 px-4 font-semibold">Ativo / Equipamento</th>
                  <th className="py-3 px-4 font-semibold">Categoria</th>
                  <th className="py-3 px-4 font-semibold">Centro de Custo</th>
                  <th className="py-3 px-4 font-semibold text-center">Criticidade</th>
                  <th className="py-3 px-4 font-semibold text-center">Estado</th>
                  <th className="py-3 px-4 font-semibold text-center">Ações</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {filteredAssets.map((asset) => {
                  const tag = asset.code_tag || asset.codeTag || 'AUTO-TAG';
                  const level = asset.criticality_level || asset.criticalityLevel || 'B';
                  return (
                    <tr key={asset.id} className="hover:bg-slate-50/80 transition-colors">
                      <td className="py-3 px-4 font-mono font-bold text-blue-700">{tag}</td>
                      <td className="py-3 px-4">
                        <div className="flex flex-col">
                          <span className="text-slate-900 font-medium text-sm">{asset.name}</span>
                          <span className="text-xs text-slate-500 mt-0.5">{asset.brand} {asset.model}</span>
                        </div>
                      </td>
                      <td className="py-3 px-4 text-xs text-slate-600">
                        {formatCategory(asset.category)}
                      </td>
                      <td className="py-3 px-4 text-xs text-slate-600">
                        {asset.cost_center || asset.costCenter}
                      </td>
                      <td className="py-3 px-4 text-center">
                        <span className={`inline-flex px-2 py-0.5 rounded-md text-[11px] font-medium border ${
                          level === 'A' 
                            ? 'bg-red-50 text-red-700 border-red-200' 
                            : level === 'B' 
                            ? 'bg-amber-50 text-amber-700 border-amber-200' 
                            : 'bg-blue-50 text-blue-700 border-blue-200'
                        }`}>
                          Nível {level}
                        </span>
                      </td>
                      <td className="py-3 px-4 text-center">
                        {renderStatusBadge(asset.status)}
                      </td>
                      <td className="py-3 px-4 text-center">
                        <div className="flex items-center justify-center gap-1.5">
                          {/* Gestão Documental */}
                          <button
                            onClick={() => {
                              setSelectedDocAsset(asset);
                              setIsDocModalOpen(true);
                              loadDocuments(asset.id);
                            }}
                            className="p-1.5 text-slate-500 hover:text-slate-800 hover:bg-slate-100 rounded-md transition-colors"
                            title="Documentos do Ativo"
                          >
                            <FileText size={15} />
                          </button>
                          
                          {/* QR Code */}
                          <button
                            onClick={() => {
                              setSelectedQrAsset(asset);
                              setIsQrModalOpen(true);
                            }}
                            className="p-1.5 text-slate-500 hover:text-slate-800 hover:bg-slate-100 rounded-md transition-colors"
                            title="Código QR / Etiqueta"
                          >
                            <QrCode size={15} />
                          </button>

                          {/* Editar */}
                          {!isTechnician && !isDriverOperator && (
                            <button
                              onClick={() => handleOpenEditModal(asset)}
                              className="p-1.5 text-slate-500 hover:text-slate-800 hover:bg-slate-100 rounded-md transition-colors"
                              title="Editar Ativo"
                            >
                              <Edit2 size={15} />
                            </button>
                          )}

                          {/* Arquivar / Delete */}
                          {!isTechnician && !isDriverOperator && (
                            <button
                              onClick={() => {
                                setDeletingAsset(asset);
                                setIsDeleteModalOpen(true);
                              }}
                              className="p-1.5 text-red-600 hover:text-red-700 hover:bg-red-50 rounded-md transition-colors"
                              title="Arquivar Ativo"
                            >
                              <Trash2 size={15} />
                            </button>
                          )}
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        )}
      </div>

      {/* ==========================================
          MODAL: ASSISTENTE DE CADASTRO (4 ETAPAS)
          ========================================== */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-xl rounded-md shadow-xl overflow-hidden flex flex-col">
            
            {/* Header */}
            <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-slate-50">
              <div className="flex items-center gap-2">
                <Wrench className="text-blue-700" size={18} />
                <h2 className="text-base font-bold text-slate-900">
                  {modalMode === 'create' ? 'Cadastrar Novo Ativo' : 'Editar Ativo Corporativo'}
                </h2>
              </div>
              <button
                onClick={() => setIsModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            {/* Indicador de Passos */}
            <div className="bg-slate-50/60 px-6 py-3 border-b border-slate-200 flex items-center justify-between text-xs font-semibold">
              <div className="flex items-center gap-2">
                <span className={`w-5 h-5 rounded-full flex items-center justify-center border font-mono text-[11px] ${
                  currentStep >= 1 ? 'bg-blue-50 border-blue-600 text-blue-700 font-bold' : 'border-slate-300 text-slate-400'
                }`}>1</span>
                <span className={currentStep >= 1 ? 'text-slate-900' : 'text-slate-400'}>Identificação</span>
              </div>
              <ChevronRight size={14} className="text-slate-400" />
              <div className="flex items-center gap-2">
                <span className={`w-5 h-5 rounded-full flex items-center justify-center border font-mono text-[11px] ${
                  currentStep >= 2 ? 'bg-blue-50 border-blue-600 text-blue-700 font-bold' : 'border-slate-300 text-slate-400'
                }`}>2</span>
                <span className={currentStep >= 2 ? 'text-slate-900' : 'text-slate-400'}>Operação</span>
              </div>
              <ChevronRight size={14} className="text-slate-400" />
              <div className="flex items-center gap-2">
                <span className={`w-5 h-5 rounded-full flex items-center justify-center border font-mono text-[11px] ${
                  currentStep >= 3 ? 'bg-blue-50 border-blue-600 text-blue-700 font-bold' : 'border-slate-300 text-slate-400'
                }`}>3</span>
                <span className={currentStep >= 3 ? 'text-slate-900' : 'text-slate-400'}>Conformidade</span>
              </div>
              {modalMode === 'create' && (
                <>
                  <ChevronRight size={14} className="text-slate-400" />
                  <div className="flex items-center gap-2">
                    <span className={`w-5 h-5 rounded-full flex items-center justify-center border font-mono text-[11px] ${
                      currentStep >= 4 ? 'bg-blue-50 border-blue-600 text-blue-700 font-bold' : 'border-slate-300 text-slate-400'
                    }`}>4</span>
                    <span className={currentStep >= 4 ? 'text-slate-900' : 'text-slate-400'}>Anexos</span>
                  </div>
                </>
              )}
            </div>

            {/* Formulário Wizard */}
            <form onSubmit={handleSubmitForm} className="p-6 space-y-4 flex-1 overflow-y-auto max-h-[60vh] scrollbar-thin">
              
              {/* ETAPA 1: IDENTIFICAÇÃO */}
              {currentStep === 1 && (
                <div className="space-y-4">
                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                        Categoria de Ativo *
                      </label>
                      <select
                        value={category}
                        disabled={modalMode === 'edit'} // Categoria fixa na edição
                        onChange={(e) => setCategory(e.target.value as any)}
                        className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none cursor-pointer"
                      >
                        {categories.map((c) => (
                          <option key={c.value} value={c.value}>
                            {c.label}
                          </option>
                        ))}
                      </select>
                    </div>
                    <div>
                      <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                        Tag / Código do Ativo
                      </label>
                      <input
                        type="text"
                        value={codeTag}
                        onChange={(e) => setCodeTag(e.target.value)}
                        placeholder="Deixe vazio para gerar automaticamente"
                        className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                      />
                    </div>
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                      Nome / Designação do Ativo *
                    </label>
                    <input
                      type="text"
                      required
                      value={name}
                      onChange={(e) => setName(e.target.value)}
                      placeholder={domain === 'fleet' ? "ex: Camião Volvo FMX Basculante" : "ex: Gerador Cummins 500kVA ou Extrusora N2"}
                      className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"

                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">

                        value={costCenter}

                    </div>
                    <div>
                      <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">

                        value={criticalityLevel}
                        onChange={(e) => setCriticalityLevel(e.target.value as any)}

                      >
                        <option value="A">Nível A (Alta Paragem Crítica)</option>
                        <option value="B">Nível B (Média Criticidade)</option>
                        <option value="C">Nível C (Baixa Criticidade)</option>
                      </select>
                    </div>
                  </div>
                </div>
              )}

              {/* ETAPA 2: OPERAÇÃO (Condicional) */}
              {currentStep === 2 && (
                <div className="space-y-4">
                  {domain === 'fleet' ? (
                    /* CAMPOS PARA VEÍCULOS (FROTA) */
                    <>
                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Matrícula / Registro *
                          </label>
                          <input
                            type="text"
                            required
                            value={licensePlate}
                            onChange={(e) => setLicensePlate(e.target.value)}
                            placeholder="ex: MM-10-24"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Número de Chassis (VIN)
                          </label>
                          <input
                            type="text"
                            value={chassisVin}
                            onChange={(e) => setChassisVin(e.target.value)}
                            placeholder="ex: 938Y498HF8..."
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>

                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Número do Motor
                          </label>
                          <input
                            type="text"
                            value={engineNumber}
                            onChange={(e) => setEngineNumber(e.target.value)}
                            placeholder="ex: D13K460"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Capacidade do Depósito (Litros)
                          </label>
                          <input
                            type="number"
                            value={fuelTankCapacityLiters}
                            onChange={(e) => setFuelTankCapacityLiters(e.target.value !== '' ? Number(e.target.value) : '')}
                            placeholder="ex: 300"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>

                      <div>
                        <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                          {counterLabel} *
                        </label>
                        <input
                          type="number"
                          required
                          value={currentMeter}
                          onChange={(e) => setCurrentMeter(Number(e.target.value))}
                          className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                        />
                      </div>
                    </>
                  ) : (
                    /* CAMPOS PARA GERADORES E MAQUINARIA (FABRIL) */
                    <>
                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Setor Fabril / Área *
                          </label>
                          <input
                            type="text"
                            required
                            value={plantSector}
                            onChange={(e) => setPlantSector(e.target.value)}
                            placeholder="ex: Moagem, Embalamento, Linha 2"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Especificações Elétricas (Voltagem)
                          </label>
                          <input
                            type="text"
                            value={voltageSpecs}
                            onChange={(e) => setVoltageSpecs(e.target.value)}
                            placeholder="ex: 380V / 3 Fases"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>

                      <div>
                        <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                          Potência Estimada (kW / kVA)
                        </label>
                        <input
                          type="number"
                          value={powerKwKva}
                          onChange={(e) => setPowerKwKva(e.target.value !== '' ? Number(e.target.value) : '')}
                          placeholder="ex: 150"
                          className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                        />
                      </div>

                      <div>
                        <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                          {counterLabel} *
                        </label>
                        <input
                          type="number"
                          required
                          value={currentMeter}
                          onChange={(e) => setCurrentMeter(Number(e.target.value))}
                          className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                        />
                      </div>
                    </>
                  )}
                </div>
              )}

              {/* ETAPA 3: CONFORMIDADE */}
              {currentStep === 3 && (
                <div className="space-y-4">
                  {domain === 'fleet' ? (
                    /* CONFORMIDADE DE TRÂNSITO INATRO (FROTA) */
                    <>
                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Validade Inspeção INATRO
                          </label>
                          <input
                            type="date"
                            value={inatroInspectionExpiry}
                            onChange={(e) => setInatroInspectionExpiry(e.target.value)}
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Companhia de Seguros
                          </label>
                          <input
                            type="text"
                            value={insuranceCompany}
                            onChange={(e) => setInsuranceCompany(e.target.value)}
                            placeholder="ex: EMOSE, Fidelidade"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                          />
                        </div>
                      </div>

                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Número de Apólice do Seguro
                          </label>
                          <input
                            type="text"
                            value={insurancePolicyNumber}
                            onChange={(e) => setInsurancePolicyNumber(e.target.value)}
                            placeholder="ex: POL-8429-01"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Validade do Seguro
                          </label>
                          <input
                            type="date"
                            value={insuranceExpiry}
                            onChange={(e) => setInsuranceExpiry(e.target.value)}
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>
                    </>
                  ) : (
                    /* CONFORMIDADE INDUSTRIAL INNOQ / PRESSÃO (FABRIL) */
                    <>
                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Validade Calibração INNOQ
                          </label>
                          <input
                            type="date"
                            value={innoqCalibrationExpiry}
                            onChange={(e) => setInnoqCalibrationExpiry(e.target.value)}
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Validade Teste Pressão/Caldeira
                          </label>
                          <input
                            type="date"
                            value={pressureTestExpiry}
                            onChange={(e) => setPressureTestExpiry(e.target.value)}
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>

                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Fornecedor Local Moçambicano
                          </label>
                          <input
                            type="text"
                            value={localSupplierName}
                            onChange={(e) => setLocalSupplierName(e.target.value)}
                            placeholder="ex: Barloworld Moçambique"
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                          />
                        </div>
                        <div>
                          <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                            Fim da Garantia
                          </label>
                          <input
                            type="date"
                            value={warrantyExpiry}
                            onChange={(e) => setWarrantyExpiry(e.target.value)}
                            className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                          />
                        </div>
                      </div>
                    </>
                  )}
                </div>
              )}

              {/* ETAPA 4: ANEXOS DOCUMENTAIS (Apenas na criação) */}
              {currentStep === 4 && modalMode === 'create' && (
                <div className="space-y-4">
                  <span className="text-xs font-semibold text-slate-600 block mb-2 uppercase tracking-wide">
                    Carregar Ficha Técnica Inicial (PDF, PNG ou JPEG até 20MB)
                  </span>

                  <div
                    onDragEnter={handleDrag}
                    onDragLeave={handleDrag}
                    onDragOver={handleDrag}
                    onDrop={handleDrop}
                    className={`border-2 border-dashed rounded-md p-8 text-center transition-all flex flex-col items-center justify-center gap-3 cursor-pointer ${
                      isDragActive 
                        ? 'border-blue-600 bg-blue-50/50' 
                        : wizardFile 
                        ? 'border-blue-500 bg-blue-50/20' 
                        : 'border-slate-300 bg-slate-50/60 hover:border-slate-400'
                    }`}
                  >
                    <input
                      type="file"
                      id="wizard-file"
                      onChange={handleFileChange}
                      accept=".pdf,.png,.jpeg,.jpg,.webp"
                      className="hidden"
                    />
                    <label htmlFor="wizard-file" className="cursor-pointer flex flex-col items-center gap-2">
                      <UploadCloud size={36} className="text-slate-400 group-hover:text-slate-600" />
                      {wizardFile ? (
                        <div className="text-xs font-medium text-blue-700">
                          Ficheiro selecionado: <span className="font-mono underline">{wizardFile.name}</span>
                          <p className="text-[10px] text-slate-500 mt-1">({(wizardFile.size / 1024 / 1024).toFixed(2)} MB)</p>
                        </div>
                      ) : (
                        <div className="text-xs text-slate-600">
                          Arrastar e largar ficheiro aqui ou <span className="text-blue-700 font-semibold underline">procurar</span>
                          <p className="text-[10px] text-slate-400 mt-1">PDF, JPEG, PNG ou WebP de até 20MB</p>
                        </div>
                      )}
                    </label>
                  </div>
                </div>
              )}

              {/* Botões do Assistente */}
              <div className="flex justify-between items-center border-t border-slate-200 pt-5 mt-2">
                <div>
                  {currentStep > 1 && (
                    <button
                      type="button"
                      onClick={handlePrevStep}
                      disabled={submitting}
                      className="px-4 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 rounded-md text-xs font-medium flex items-center gap-1.5 transition-all cursor-pointer disabled:opacity-50"
                    >
                      <ChevronLeft size={14} />
                      <span>Anterior</span>
                    </button>
                  )}
                </div>

                <div className="flex gap-2">
                  <button
                    type="button"
                    onClick={() => setIsModalOpen(false)}
                    disabled={submitting}
                    className="px-4 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 rounded-md text-xs font-medium transition-all"
                  >
                    Cancelar
                  </button>

                  {/* Verifica se deve mostrar botão "Avançar" ou "Submeter" */}
                  {((modalMode === 'create' && currentStep < 4) || (modalMode === 'edit' && currentStep < 3)) ? (
                    <button
                      type="button"
                      onClick={handleNextStep}
                      className="px-4 py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center gap-1.5 transition-all cursor-pointer shadow-none"
                    >
                      <span>Seguinte</span>
                      <ChevronRight size={14} />
                    </button>
                  ) : (
                    <button
                      type="submit"
                      disabled={submitting}
                      className="px-4 py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center gap-1.5 transition-all cursor-pointer disabled:opacity-50 shadow-none"
                    >
                      {submitting ? (
                        <>
                          <Loader2 size={14} className="animate-spin" />
                          <span>A processar...</span>
                        </>
                      ) : (
                        <span>{modalMode === 'create' ? 'Finalizar Registo' : 'Atualizar Ativo'}</span>
                      )}
                    </button>
                  )}
                </div>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: GESTÃO DOCUMENTAL DO ATIVO
          ========================================== */}
      {isDocModalOpen && selectedDocAsset && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-lg rounded-md shadow-xl overflow-hidden flex flex-col">
            
            {/* Header */}
            <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-slate-50">
              <div className="flex items-center gap-2">
                <FileText className="text-blue-700" size={18} />
                <h2 className="text-base font-bold text-slate-900">
                  Documentos: {selectedDocAsset.code_tag || selectedDocAsset.codeTag}
                </h2>
              </div>
              <button
                onClick={() => {
                  setIsDocModalOpen(false);
                  setSelectedDocAsset(null);
                  setSelectedFile(null);
                  setDocTitle('');
                  if (fileInputRef.current) {
                    fileInputRef.current.value = '';
                  }
                }}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            <div className="p-6 space-y-5 overflow-y-auto max-h-[70vh] scrollbar-thin">
              
              {/* Formulário para Envio de Novo Documento */}
              <form onSubmit={handleAddAdditionalDocument} className="bg-slate-50 p-4 border border-slate-200 rounded-md space-y-3">
                <span className="text-xs font-bold text-blue-700 uppercase tracking-wider block mb-1">
                  Adicionar Anexo de Manutenção / Conformidade
                </span>
                
                <div>
                  <label className="block text-[11px] font-semibold text-slate-600 mb-1 uppercase">Título *</label>
                  <input
                    type="text"
                    required
                    value={docTitle}
                    onChange={(e) => setDocTitle(e.target.value)}
                    placeholder="ex: Manual do Operador, Apólice de Seguro 2026"
                    className="w-full bg-white border border-slate-300 rounded-md px-3 py-1.5 text-xs text-slate-900 focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                  />
                </div>

                <div className="grid grid-cols-2 gap-3">
                  <div>
                    <label className="block text-[11px] font-semibold text-slate-600 mb-1 uppercase">Tipo *</label>
                    <select
                      value={docType}
                      onChange={(e) => setDocType(e.target.value)}
                      className="w-full bg-white border border-slate-300 rounded-md px-2.5 py-1.5 text-xs text-slate-900 focus:outline-none focus:border-blue-600 focus:ring-1 focus:ring-blue-600"
                    >
                      <option value="tech_sheet">Ficha Técnica</option>
                      <option value="manual">Manual Técnico</option>
                      <option value="diagram">Diagrama Elétrico / Mecânico</option>
                      <option value="inspection_certificate">Certificado de Inspeção (INATRO/INNOQ)</option>
                      <option value="insurance">Apólice de Seguro</option>
                      <option value="asset_photo">Fotografia do Ativo</option>
                    </select>
                  </div>
                  <div>
                    <label className="block text-[11px] font-semibold text-slate-600 mb-1 uppercase">Ficheiro (20MB max) *</label>
                    <input
                      type="file"
                      ref={fileInputRef}
                      required
                      accept=".pdf,.png,.jpeg,.jpg,.webp"
                      onChange={(e) => setSelectedFile(e.target.files?.[0] || null)}
                      className="w-full text-xs text-slate-500 file:mr-2 file:py-1 file:px-2 file:rounded-md file:border-0 file:text-[10px] file:font-semibold file:bg-slate-200 file:text-slate-700 file:cursor-pointer"
                    />
                  </div>
                </div>

                <button
                  type="submit"
                  disabled={isUploadingDoc || !selectedFile}
                  className="w-full py-2 bg-blue-700 hover:bg-blue-800 text-white text-xs font-medium rounded-md flex items-center justify-center gap-1.5 transition-all cursor-pointer disabled:opacity-50 shadow-none"
                >
                  {isUploadingDoc ? (
                    <>
                      <Loader2 size={13} className="animate-spin" />
                      <span>A carregar ficheiro...</span>
                    </>
                  ) : (
                    <>
                      <UploadCloud size={13} />
                      <span>Anexar Documento</span>
                    </>
                  )}
                </button>
              </form>

              {/* Lista de Documentos Atuais */}
              <div className="space-y-3">
                <span className="text-xs font-bold text-slate-700 uppercase tracking-wider block">
                  Documentação Arquivada ({documents.length})
                </span>

                {docLoading ? (
                  <div className="text-center py-6">
                    <Loader2 size={24} className="animate-spin text-slate-400 mx-auto" />
                  </div>
                ) : documents.length === 0 ? (
                  <div className="text-center py-6 border border-slate-200 border-dashed rounded-md bg-slate-50/50">
                    <p className="text-xs text-slate-500">Nenhum documento anexado a este ativo.</p>
                  </div>
                ) : (
                  <div className="space-y-2">
                    {documents.map((doc) => {
                      const url = doc.signed_url || doc.signedUrl || '#';
                      return (
                        <div key={doc.id} className="p-3 bg-white border border-slate-200 rounded-md flex items-center justify-between text-xs hover:border-slate-300 transition-colors">
                          <div className="flex items-center gap-2 flex-1 truncate mr-4">
                            <File size={16} className="text-blue-700 shrink-0" />
                            <div className="truncate">
                              <span className="font-semibold text-slate-800 block truncate" title={doc.title}>
                                {doc.title}
                              </span>
                              <span className="text-[10px] text-slate-500 block mt-0.5">
                                {formatDocType(doc.document_type || doc.documentType || '')} · {doc.created_at ? new Date(doc.created_at).toLocaleDateString('pt-MZ') : '---'}
                              </span>
                            </div>
                          </div>

                          {/* Abrir URL assinada */}
                          <a
                            href={url}
                            target="_blank"
                            rel="noopener noreferrer"
                            className="px-2.5 py-1.5 bg-white hover:bg-slate-50 border border-slate-300 text-slate-700 rounded-md text-[10px] font-medium flex items-center gap-1 shrink-0"
                          >
                            <Eye size={12} />
                            <span>Abrir</span>
                          </a>
                        </div>
                      );
                    })}
                  </div>
                )}
              </div>

            </div>
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: VISUALIZAÇÃO E IMPRESSÃO DE QR CODE
          ========================================== */}
      {isQrModalOpen && selectedQrAsset && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in print:bg-white print:text-black">
          <div className="bg-white border border-slate-200 w-full max-w-sm rounded-md shadow-xl overflow-hidden flex flex-col print:border-none print:shadow-none print:bg-white">
            
            {/* Header (Escondido na impressão) */}
            <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-slate-50 print:hidden">
              <div className="flex items-center gap-2">
                <QrCode className="text-blue-700" size={18} />
                <h2 className="text-base font-bold text-slate-900">Etiqueta QR Code</h2>
              </div>
              <button
                onClick={() => {
                  setIsQrModalOpen(false);
                  setSelectedQrAsset(null);
                }}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            {/* Corpo / Etiqueta Imprimível */}
            <div className="p-6 flex flex-col items-center justify-center text-center space-y-4 print:p-0 print:text-black">
              
              {/* Cartão de Etiqueta com Borda Refinada */}
              <div className="bg-white p-5 rounded-md border border-slate-200 flex flex-col items-center shadow-sm w-full max-w-[260px] print:shadow-none print:border-2 print:border-black">
                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-widest mb-2 print:text-black">AVEXAS CMMS</span>
                
                {/* Canvas do QR Code */}
                <canvas ref={qrCanvasRef} className="mx-auto" />
                
                <div className="mt-3 text-slate-900 print:text-black">
                  <span className="font-mono font-bold text-sm tracking-wide block uppercase text-slate-900">
                    {selectedQrAsset.code_tag || selectedQrAsset.codeTag}
                  </span>
                  <span className="text-xs font-semibold block text-slate-700 truncate max-w-[200px] mt-0.5">
                    {selectedQrAsset.name}
                  </span>
                  <span className="text-[10px] text-slate-500 block font-medium mt-0.5">
                    {formatCategory(selectedQrAsset.category)}
                  </span>
                </div>
              </div>

              {/* Botões de Ação (Escondidos na impressão) */}
              <div className="flex gap-2 w-full pt-4 border-t border-slate-200 print:hidden">
                <button
                  onClick={handleDownloadQrCode}
                  className="flex-1 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all cursor-pointer"
                >
                  <Download size={14} />
                  <span>PNG</span>
                </button>
                <button
                  onClick={handlePrintLabel}
                  className="flex-1 py-2 bg-blue-700 hover:bg-blue-800 text-white rounded-md text-xs font-medium flex items-center justify-center gap-1.5 transition-all cursor-pointer shadow-none"
                >
                  <Printer size={14} />
                  <span>Imprimir</span>
                </button>
              </div>
            </div>

          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: CONFIRMAÇÃO DE ARQUIVAMENTO (DELETE)
          ========================================== */}
      {isDeleteModalOpen && deletingAsset && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-sm rounded-md shadow-xl p-6 space-y-4">
            <div className="w-12 h-12 bg-red-50 text-red-600 border border-red-200 rounded-md flex items-center justify-center">
              <AlertTriangle size={24} />
            </div>
            <div>
              <h3 className="text-base font-bold text-slate-900">Arquivar Ativo de Manutenção</h3>
              <p className="text-xs text-slate-600 mt-1.5 leading-relaxed">
                Tem a certeza que deseja arquivar o ativo <span className="font-bold text-slate-900 font-mono">{deletingAsset.code_tag || deletingAsset.codeTag}</span>? 
                Esta ação moverá o equipamento para o histórico e o impedirá de abrir novas ordens de serviço.
              </p>
            </div>
            <div className="flex justify-end gap-3 pt-2">
              <button
                onClick={() => {
                  setIsDeleteModalOpen(false);
                  setDeletingAsset(null);
                }}
                className="px-4 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 text-xs font-medium rounded-md transition-all"
              >
                Cancelar
              </button>
              <button
                onClick={handleDeleteAsset}
                className="px-4 py-2 bg-red-600 hover:bg-red-700 active:bg-red-800 text-white text-xs font-medium rounded-md transition-all shadow-none"
              >
                Confirmar Arquivamento
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ==========================================
          MODAL: IMPORTAÇÃO EM LOTE (CSV BATCH IMPORT)
          ========================================== */}
      {isBatchImportModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-2xl rounded-md shadow-xl overflow-hidden flex flex-col max-h-[90vh]">
            
            {/* Cabeçalho do Modal */}
            <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-slate-50">
              <div className="flex items-center gap-2">
                <FileSpreadsheet className="text-blue-700" size={18} />
                <h3 className="text-sm font-bold text-slate-900">Importação de Ativos em Lote (CSV)</h3>
              </div>
              <button
                onClick={() => setIsBatchImportModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            {/* Conteúdo do Modal (Rolável) */}
            <div className="p-6 space-y-5 overflow-y-auto text-xs">
              
              {/* Seção 1: Instruções e Modelo */}
              <div className="bg-slate-50 border border-slate-200 rounded-md p-4 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
                <div>
                  <span className="font-semibold text-slate-800 block">Modelo de Ficheiro CSV para {currentScope === 'plant' ? 'Equipamentos Fabris' : 'Frota Rodoviária'}</span>
                  <p className="text-slate-500 text-[11px] mt-0.5">
                    {currentScope === 'plant'
                      ? 'O modelo contém colunas para Nome, Categoria (generator/machinery), Tag, Fabricante, Modelo, Ano, Setor, Potência, Voltagem e Horímetro.'
                      : 'O modelo contém colunas para Nome, Categoria (vehicle_light/vehicle_heavy), Tag, Marca, Modelo, Ano, Matrícula, Chassis e Odómetro.'}
                  </p>
                </div>
                <button
                  type="button"
                  onClick={handleDownloadCsvTemplate}
                  className="bg-white hover:bg-slate-100 border border-slate-300 text-slate-700 font-medium py-2 px-3 rounded-md flex items-center gap-1.5 shrink-0 transition-colors cursor-pointer"
                >
                  <Download size={14} className="text-slate-600" />
                  <span>Descarregar Modelo CSV</span>
                </button>
              </div>

              {/* Seção 2: Área Drag & Drop / Input de Ficheiro */}
              <div>
                <label className="block text-xs font-semibold text-slate-700 mb-1.5 uppercase tracking-wide">
                  Carregar Ficheiro CSV *
                </label>
                <input
                  type="file"
                  accept=".csv,text/csv,application/vnd.ms-excel"
                  ref={batchFileInputRef}
                  onChange={(e) => e.target.files && e.target.files[0] && handleBatchFileSelect(e.target.files[0])}
                  className="hidden"
                />
                <div
                  onDragOver={handleBatchDragOver}
                  onDragLeave={handleBatchDragLeave}
                  onDrop={handleBatchDrop}
                  onClick={() => batchFileInputRef.current?.click()}
                  className={`border-2 border-dashed rounded-md p-6 text-center cursor-pointer transition-colors flex flex-col items-center justify-center gap-2 ${
                    batchDragActive
                      ? 'border-blue-600 bg-blue-50/50'
                      : batchFile
                      ? 'border-emerald-300 bg-emerald-50/30'
                      : 'border-slate-300 hover:border-blue-500 bg-slate-50/50'
                  }`}
                >
                  <UploadCloud size={28} className={batchFile ? "text-emerald-600" : "text-slate-400"} />
                  {batchFile ? (
                    <div>
                      <p className="font-semibold text-slate-900">{batchFile.name}</p>
                      <p className="text-[11px] text-slate-500 font-mono mt-0.5">{(batchFile.size / 1024).toFixed(1)} KB · Clique para alterar</p>
                    </div>
                  ) : (
                    <div>
                      <p className="font-semibold text-slate-700">Arraste e solte o seu ficheiro CSV aqui ou <span className="text-blue-700 underline">procure no computador</span></p>
                      <p className="text-[11px] text-slate-400 mt-1">Apenas ficheiros .csv com codificação UTF-8</p>
                    </div>
                  )}
                </div>
              </div>

              {/* Exibição de Mensagem de Erro do Backend / Validação sem fechar o modal */}
              {batchError && (
                <div className="p-3.5 bg-red-50 border border-red-200 text-red-700 rounded-md flex gap-2.5 items-start animate-fade-in">
                  <AlertTriangle size={16} className="text-red-600 shrink-0 mt-0.5" />
                  <div className="flex-1">
                    <span className="font-bold block text-red-900">Erro na Validação / Importação</span>
                    <span className="text-[11px] leading-relaxed block text-red-700 mt-0.5">{batchError}</span>
                  </div>
                </div>
              )}

              {/* Seção 3: Pré-Visualização das Primeiras Linhas */}
              {parsedBatchData.length > 0 && (
                <div className="space-y-2">
                  <div className="flex items-center justify-between">
                    <span className="font-bold text-slate-800 uppercase tracking-wide text-[11px]">
                      Pré-Visualização dos Dados ({Math.min(5, parsedBatchData.length)} de {parsedBatchData.length} registos)
                    </span>
                    <span className="text-emerald-700 font-semibold text-[11px] bg-emerald-50 px-2 py-0.5 rounded border border-emerald-200">
                      {parsedBatchData.length} {parsedBatchData.length === 1 ? 'ativo pronto' : 'ativos prontos'}
                    </span>
                  </div>

                  <div className="border border-slate-200 rounded-md overflow-x-auto max-h-48">
                    <table className="w-full text-left text-[11px] text-slate-700">
                      <thead className="bg-slate-100 text-slate-700 font-semibold border-b border-slate-200 sticky top-0">
                        <tr>
                          <th className="px-3 py-2 border-r border-slate-200">#</th>
                          {Object.keys(parsedBatchData[0] || {}).map((col) => (
                            <th key={col} className="px-3 py-2 border-r border-slate-200 whitespace-nowrap">{col}</th>
                          ))}
                        </tr>
                      </thead>
                      <tbody className="divide-y divide-slate-200 bg-white">
                        {parsedBatchData.slice(0, 5).map((row, idx) => (
                          <tr key={idx} className="hover:bg-slate-50 font-mono">
                            <td className="px-3 py-1.5 border-r border-slate-200 font-bold text-slate-500">{idx + 1}</td>
                            {Object.keys(parsedBatchData[0] || {}).map((col) => (
                              <td key={col} className="px-3 py-1.5 border-r border-slate-200 whitespace-nowrap truncate max-w-[150px]">
                                {row[col] || <span className="text-slate-300 italic">-</span>}
                              </td>
                            ))}
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>
              )}

            </div>

            {/* Rodapé com Botões de Ação */}
            <div className="px-6 py-4 bg-slate-50 border-t border-slate-200 flex justify-end gap-3">
              <button
                type="button"
                onClick={() => {
                  setIsBatchImportModalOpen(false);
                  setBatchFile(null);
                  setParsedBatchData([]);
                  setBatchError(null);
                }}
                className="px-4 py-2 bg-white hover:bg-slate-100 text-slate-700 border border-slate-300 rounded-md text-xs font-medium transition-colors cursor-pointer"
              >
                Cancelar
              </button>
              <button
                type="button"
                onClick={handleBatchSubmit}
                disabled={batchSubmitting || parsedBatchData.length === 0}
                className="px-5 py-2 bg-blue-700 hover:bg-blue-800 disabled:opacity-50 disabled:cursor-not-allowed text-white rounded-md text-xs font-medium transition-colors shadow-none flex items-center gap-2 cursor-pointer"
              >
                {batchSubmitting ? (
                  <>
                    <Loader2 size={14} className="animate-spin" />
                    <span>A Importar...</span>
                  </>
                ) : (
                  <>
                    <FileSpreadsheet size={14} />
                    <span>Importar {parsedBatchData.length > 0 ? `${parsedBatchData.length} Ativos` : 'Ativos'}</span>
                  </>
                )}
              </button>
            </div>

          </div>
        </div>
      )}

    </div>
  );
};

// ==========================================
// PORTAL DO MOTORISTA / CONDUTOR / OPERADOR
// ==========================================
interface DriverOperatorPortalProps {
  assets: Asset[];
  fetchAssets: () => Promise<void>;
  loading: boolean;
}

const DriverOperatorPortal: React.FC<DriverOperatorPortalProps> = ({ assets, fetchAssets, loading }) => {
  const { user } = useAuth();
  const [selectedAssetId, setSelectedAssetId] = useState<string>('');
  
  // Checklist State
  const [fluids, setFluids] = useState(false);
  const [brakes, setBrakes] = useState(false);
  const [lights, setLights] = useState(false);
  const [tires, setTires] = useState(false);
  const [safety, setSafety] = useState(false);
  const [checklistSuccess, setChecklistSuccess] = useState(false);

  // Anomaly State
  const [isAnomalyModalOpen, setIsAnomalyModalOpen] = useState(false);
  const [anomalyTitle, setAnomalyTitle] = useState('');
  const [anomalyDesc, setAnomalyDesc] = useState('');
  const [anomalyPriority, setAnomalyPriority] = useState<'low' | 'medium' | 'high' | 'urgent'>('medium');
  const [currentOdometer, setCurrentOdometer] = useState<number | ''>('');
  const [submittingAnomaly, setSubmittingAnomaly] = useState(false);
  const [anomalySuccess, setAnomalySuccess] = useState(false);

  // Filter out non-vehicles for drivers
  const vehicles = assets.filter(a => a.category === 'vehicle_light' || a.category === 'vehicle_heavy' || a.category?.startsWith('vehicle'));

  useEffect(() => {
    if (vehicles.length > 0 && !selectedAssetId) {
      setSelectedAssetId(vehicles[0].id);
      const m = vehicles[0].current_meter ?? vehicles[0].currentMeter ?? '';
      setCurrentOdometer(m);
    }
  }, [vehicles]);

  const handleSelectAsset = (id: string) => {
    setSelectedAssetId(id);
    const selected = vehicles.find(v => v.id === id);
    if (selected) {
      const m = selected.current_meter ?? selected.currentMeter ?? '';
      setCurrentOdometer(m);
    }
    // reset checklist
    setFluids(false);
    setBrakes(false);
    setLights(false);
    setTires(false);
    setSafety(false);
    setChecklistSuccess(false);
  };

  const handleChecklistSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedAssetId) return;
    setChecklistSuccess(true);
    setTimeout(() => setChecklistSuccess(false), 5000);
  };

  const handleAnomalySubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedAssetId || !anomalyTitle.trim() || !anomalyDesc.trim()) {
      alert('Por favor, preencha todos os campos obrigatórios.');
      return;
    }

    setSubmittingAnomaly(true);
    const payload = {
      assetId: selectedAssetId,
      asset_id: selectedAssetId,
      title: anomalyTitle,
      description: anomalyDesc,
      orderType: 'corrective',
      order_type: 'corrective',
      priority: anomalyPriority,
      meterAtService: currentOdometer !== '' ? Number(currentOdometer) : undefined,
      meter_at_service: currentOdometer !== '' ? Number(currentOdometer) : undefined,
      status: 'open'
    };

    try {
      await api.post('/work-orders', payload);
      setAnomalySuccess(true);
      setAnomalyTitle('');
      setAnomalyDesc('');
      setTimeout(() => {
        setAnomalySuccess(false);
        setIsAnomalyModalOpen(false);
      }, 2500);
    } catch (err: any) {
      console.error('Erro ao submeter anomalia:', err);
      alert(err.response?.data?.message || err.message || 'Erro ao submeter anomalia.');
    } finally {
      setSubmittingAnomaly(false);
    }
  };

  const selectedVehicle = vehicles.find(v => v.id === selectedAssetId);

  return (
    <div className="space-y-6 max-w-2xl mx-auto pb-10">
      
      {/* CARD BOAS-VINDAS */}
      <div className="bg-white border border-slate-200 rounded-md p-6 relative overflow-hidden shadow-sm animate-fade-in">
        <div className="flex items-center gap-4">
          <div className="w-12 h-12 bg-blue-50 text-blue-700 border border-blue-200 rounded-md flex items-center justify-center font-bold text-xl uppercase shrink-0">
            {user?.fullName.substring(0, 2).toUpperCase() || 'OP'}
          </div>
          <div>
            <span className="text-[10px] text-blue-700 font-bold uppercase tracking-wider block">CMMS Moçambique · Portal do Condutor</span>
            <h1 className="text-lg font-bold text-slate-900 mt-0.5">Olá, {user?.fullName}!</h1>
            <p className="text-slate-600 text-xs">Efetue vistorias de segurança e reporte avarias de campo rapidamente.</p>
          </div>
        </div>
      </div>

      {loading ? (
        <div className="bg-white border border-slate-200 rounded-md p-12 text-center">
          <RefreshCw className="animate-spin text-blue-700 mx-auto" size={24} />
          <p className="text-xs text-slate-500 mt-2 font-medium">A carregar viaturas corporativas...</p>
        </div>
      ) : vehicles.length === 0 ? (
        <div className="bg-white border border-slate-200 rounded-md p-8 text-center text-slate-500 shadow-sm">
          <AlertTriangle className="text-amber-500 mx-auto mb-2" size={24} />
          <p className="text-xs font-semibold text-slate-800">Nenhuma viatura disponível para o seu utilizador.</p>
          <p className="text-[11px] text-slate-500 mt-1">Contacte o supervisor ou gestor para associar viaturas ao seu escopo.</p>
        </div>
      ) : (
        <div className="space-y-6">
          
          {/* SELETOR DE VIATURA */}
          <div className="bg-white border border-slate-200 rounded-md p-5 space-y-3 shadow-sm">
            <label className="block text-xs font-bold text-slate-700 uppercase tracking-wide">
              Selecione a Viatura em Serviço:
            </label>
            <select
              value={selectedAssetId}
              onChange={(e) => handleSelectAsset(e.target.value)}
              className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2.5 text-xs text-slate-900 focus:outline-none"
            >
              {vehicles.map(v => (
                <option key={v.id} value={v.id}>
                  {v.code_tag || v.codeTag} - {v.name} ({v.license_plate || v.licensePlate || 'Sem Matrícula'})
                </option>
              ))}
            </select>

            {selectedVehicle && (
              <div className="mt-3 bg-slate-50 rounded-md p-4 border border-slate-200 grid grid-cols-2 gap-4 text-xs">
                <div>
                  <span className="text-slate-500 block">Marca/Modelo:</span>
                  <span className="text-slate-800 font-semibold block mt-0.5">{selectedVehicle.brand} {selectedVehicle.model}</span>
                </div>
                <div>
                  <span className="text-slate-500 block">Matrícula:</span>
                  <span className="text-slate-900 font-mono font-semibold block mt-0.5">{selectedVehicle.license_plate || selectedVehicle.licensePlate || 'N/D'}</span>
                </div>
                <div>
                  <span className="text-slate-500 block">Contador Atual:</span>
                  <span className="text-slate-800 font-semibold block mt-0.5">{selectedVehicle.current_meter ?? selectedVehicle.currentMeter ?? 0} km</span>
                </div>
                <div>
                  <span className="text-slate-500 block">Estado Operacional:</span>
                  <span className="inline-flex mt-1 items-center px-2 py-0.5 text-[10px] font-semibold bg-emerald-50 text-emerald-700 border border-emerald-200 rounded">
                    {selectedVehicle.status === 'operational' ? 'Operacional' : selectedVehicle.status === 'breakdown' ? 'Avariado' : 'Em Oficina'}
                  </span>
                </div>
              </div>
            )}
          </div>

          {/* CHECKLIST DIÁRIA DE SEGURANÇA */}
          <div className="bg-white border border-slate-200 rounded-md p-5 space-y-4 shadow-sm relative">
            <div>
              <h2 className="text-xs font-bold text-slate-900 flex items-center gap-2">
                <CheckCircle2 className="text-blue-700 shrink-0" size={16} />
                Inspeção Diária de Segurança (Checklist)
              </h2>
              <p className="text-[11px] text-slate-500 mt-0.5">Marque todos os pontos regulamentares antes de iniciar o serviço.</p>
            </div>

            {checklistSuccess ? (
              <div className="bg-emerald-50 border border-emerald-200 rounded-md p-4 text-emerald-700 text-xs text-center flex items-center justify-center gap-2 animate-fade-in">
                <CheckCircle2 size={16} />
                <span className="font-bold">Checklist guardado com sucesso! Boa viagem com segurança.</span>
              </div>
            ) : (
              <form onSubmit={handleChecklistSubmit} className="space-y-3 pt-2">
                <label className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md p-3.5 hover:bg-slate-100 transition-colors cursor-pointer group">
                  <input
                    type="checkbox"
                    checked={fluids}
                    onChange={(e) => setFluids(e.target.checked)}
                    className="accent-blue-700 bg-white border border-slate-300 rounded w-4 h-4 cursor-pointer"
                  />
                  <div className="flex-1 text-xs">
                    <span className="font-semibold text-slate-800 group-hover:text-slate-900">Fluidos & Óleos</span>
                    <p className="text-[11px] text-slate-500 mt-0.5">Níveis de óleo de motor e água de refrigeração estão corretos e sem fugas visíveis.</p>
                  </div>
                </label>

                <label className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md p-3.5 hover:bg-slate-100 transition-colors cursor-pointer group">
                  <input
                    type="checkbox"
                    checked={brakes}
                    onChange={(e) => setBrakes(e.target.checked)}
                    className="accent-blue-700 bg-white border border-slate-300 rounded w-4 h-4 cursor-pointer"
                  />
                  <div className="flex-1 text-xs">
                    <span className="font-semibold text-slate-800 group-hover:text-slate-900">Travões / Sistema de Travagem</span>
                    <p className="text-[11px] text-slate-500 mt-0.5">Travão de pé e travão de estacionamento (mão) operam perfeitamente.</p>
                  </div>
                </label>

                <label className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md p-3.5 hover:bg-slate-100 transition-colors cursor-pointer group">
                  <input
                    type="checkbox"
                    checked={lights}
                    onChange={(e) => setLights(e.target.checked)}
                    className="accent-blue-700 bg-white border border-slate-300 rounded w-4 h-4 cursor-pointer"
                  />
                  <div className="flex-1 text-xs">
                    <span className="font-semibold text-slate-800 group-hover:text-slate-900">Iluminação & Sinalização</span>
                    <p className="text-[11px] text-slate-500 mt-0.5">Faróis médios/máximos, piscas direcionais, stops e luzes de emergência limpos e funcionais.</p>
                  </div>
                </label>

                <label className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md p-3.5 hover:bg-slate-100 transition-colors cursor-pointer group">
                  <input
                    type="checkbox"
                    checked={tires}
                    onChange={(e) => setTires(e.target.checked)}
                    className="accent-blue-700 bg-white border border-slate-300 rounded w-4 h-4 cursor-pointer"
                  />
                  <div className="flex-1 text-xs">
                    <span className="font-semibold text-slate-800 group-hover:text-slate-900">Pneus & Jantes</span>
                    <p className="text-[11px] text-slate-500 mt-0.5">Rastos em condições regulamentares, pressão correta e pneu suplente disponível.</p>
                  </div>
                </label>

                <label className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md p-3.5 hover:bg-slate-100 transition-colors cursor-pointer group">
                  <input
                    type="checkbox"
                    checked={safety}
                    onChange={(e) => setSafety(e.target.checked)}
                    className="accent-blue-700 bg-white border border-slate-300 rounded w-4 h-4 cursor-pointer"
                  />
                  <div className="flex-1 text-xs">
                    <span className="font-semibold text-slate-800 group-hover:text-slate-900">Equipamentos de Proteção</span>
                    <p className="text-[11px] text-slate-500 mt-0.5">Cintos de segurança operacionais, triângulo, colete refletor e extintor de incêndio carregado.</p>
                  </div>
                </label>

                <button
                  type="submit"
                  disabled={!fluids || !brakes || !lights || !tires || !safety}
                  className="w-full bg-blue-700 hover:bg-blue-800 text-white text-xs font-medium py-3 px-4 rounded-md transition-all shadow-none mt-4 cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed"
                >
                  Validar Vistoria de Segurança
                </button>
              </form>
            )}
          </div>

          {/* BOTÃO DE ANOMALIA / AVARIA */}
          <div className="bg-red-50 border border-red-200 rounded-md p-5 space-y-3 shadow-sm">
            <div>
              <h2 className="text-xs font-bold text-red-700 flex items-center gap-2">
                <AlertTriangle className="text-red-600" size={16} />
                Reportar Avaria ou Anomalia Mecânica
              </h2>
              <p className="text-[11px] text-slate-600 mt-0.5">
                Detetou algum problema durante a condução? Abra imediatamente uma ocorrência para os técnicos.
              </p>
            </div>

            <button
              onClick={() => setIsAnomalyModalOpen(true)}
              className="w-full bg-red-600 hover:bg-red-700 active:bg-red-800 text-white text-xs font-medium py-2.5 rounded-md transition-all cursor-pointer flex items-center justify-center gap-2 shadow-none"
            >
              <AlertTriangle size={15} />
              <span>Reportar Anomalia Técnica</span>
            </button>
          </div>

        </div>
      )}

      {/* MODAL DE SUBMETER ANOMALIA */}
      {isAnomalyModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/40 backdrop-blur-xs animate-fade-in">
          <div className="bg-white border border-slate-200 w-full max-w-md rounded-md shadow-xl overflow-hidden flex flex-col relative">

            <div className="px-6 py-4 border-b border-slate-200 flex items-center justify-between bg-slate-50">
              <h3 className="text-xs font-bold text-red-700 flex items-center gap-2">
                <AlertTriangle size={16} />
                Reportar Ocorrência em Campo
              </h3>
              <button
                onClick={() => setIsAnomalyModalOpen(false)}
                className="p-1 text-slate-400 hover:text-slate-700 rounded-md hover:bg-slate-100 transition-colors cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            {anomalySuccess ? (
              <div className="p-8 text-center space-y-4">
                <div className="w-12 h-12 bg-emerald-50 text-emerald-600 rounded-md flex items-center justify-center border border-emerald-200 mx-auto">
                  <CheckCircle2 size={24} />
                </div>
                <div>
                  <h4 className="text-sm font-bold text-slate-900">Ocorrência Registada!</h4>
                  <p className="text-xs text-slate-600 mt-1">A equipa de manutenção foi notificada e uma OS Corretiva foi aberta.</p>
                </div>
              </div>
            ) : (
              <form onSubmit={handleAnomalySubmit} className="p-6 space-y-4 text-xs">
                <div>
                  <span className="text-slate-500 font-semibold block">Viatura Selecionada:</span>
                  <span className="text-slate-900 font-bold font-mono mt-1 block">
                    {selectedVehicle?.code_tag || selectedVehicle?.codeTag} - {selectedVehicle?.name}
                  </span>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                    Título / Resumo da Avaria *
                  </label>
                  <input
                    type="text"
                    required
                    value={anomalyTitle}
                    onChange={(e) => setAnomalyTitle(e.target.value)}
                    placeholder="ex: Fuga de óleo sob o radiador"
                    className="w-full bg-white border border-slate-300 focus:border-red-600 focus:ring-1 focus:ring-red-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none font-mono"
                  />
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                    Descrição Detalhada do Problema *
                  </label>
                  <textarea
                    required
                    value={anomalyDesc}
                    onChange={(e) => setAnomalyDesc(e.target.value)}
                    rows={3}
                    placeholder="Descreva o comportamento do veículo, barulhos suspeitos ou luzes do painel acesas..."
                    className="w-full bg-white border border-slate-300 focus:border-red-600 focus:ring-1 focus:ring-red-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none leading-relaxed font-sans"
                  />
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                      Quilometragem Atual *
                    </label>
                    <input
                      type="number"
                      required
                      value={currentOdometer}
                      onChange={(e) => setCurrentOdometer(e.target.value === '' ? '' : Number(e.target.value))}
                      placeholder="km atual"
                      className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold text-slate-600 mb-1.5 uppercase tracking-wide">
                      Gravidade Indicada
                    </label>
                    <select
                      value={anomalyPriority}
                      onChange={(e) => setAnomalyPriority(e.target.value as any)}
                      className="w-full bg-white border border-slate-300 focus:border-blue-600 focus:ring-1 focus:ring-blue-600 rounded-md px-3 py-2 text-xs text-slate-900 focus:outline-none"
                    >
                      <option value="low">Baixa (Pode circular)</option>
                      <option value="medium">Média (Requer atenção)</option>
                      <option value="high">Alta (Intervenção urgente)</option>
                      <option value="urgent">Crítica (Imobilizar viatura)</option>
                    </select>
                  </div>
                </div>

                <div className="flex justify-end gap-3 pt-4 border-t border-slate-200">
                  <button
                    type="button"
                    onClick={() => setIsAnomalyModalOpen(false)}
                    className="px-4 py-2 bg-white hover:bg-slate-50 text-slate-700 border border-slate-300 rounded-md text-xs font-medium transition-colors cursor-pointer"
                  >
                    Cancelar
                  </button>
                  <button
                    type="submit"
                    disabled={submittingAnomaly}
                    className="px-4 py-2 bg-red-600 hover:bg-red-700 active:bg-red-800 text-white rounded-md text-xs font-medium transition-colors shadow-none flex items-center gap-1.5 cursor-pointer"
                  >
                    {submittingAnomaly ? (
                      <>
                        <RefreshCw className="animate-spin" size={13} />
                        <span>A reportar...</span>
                      </>
                    ) : (
                      <span>Submeter Ocorrência</span>
                    )}
                  </button>
                </div>
              </form>
            )}
          </div>
        </div>
      )}

    </div>
  );
};

export default AssetsPage;
